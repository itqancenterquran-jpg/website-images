function getDashboardData_() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  return {
    students: getDataCount_(ss.getSheetByName('الطلاب')),
    teachers: getDataCount_(ss.getSheetByName('المدرسون')),
    supervisors: getDataCount_(ss.getSheetByName('المشرفون')),
    lessons: getDataCount_(ss.getSheetByName('الحصص')),
    attendance: getDataCount_(ss.getSheetByName('الحضور')),
    payments: getDataCount_(ss.getSheetByName('المدفوعات'))
  };
}


function getDataCount_(sheet) {

  if (!sheet) return 0;

  const lastRow = sheet.getLastRow();

  return lastRow > 1 ? lastRow - 1 : 0;
}


/* =========================
   الطلاب
========================= */

function getStudents_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('الطلاب');
  if (!sheet) throw new Error('ورقة الطلاب غير موجودة');

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  const n = lastRow - 1;
  const data = sheet.getRange(2, 1, n, 17).getValues();
  // الوقت بالشكل المعروض في الشيت، لتجنب مشكلة Date سنة 1899
  const timeShown = sheet.getRange(2, 12, n, 1).getDisplayValues();

  const students = [];
  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    if (String(row[0] || '').trim() === '') continue;

    students.push({
      id: String(row[0] || ''),
      name: String(row[1] || ''),
      whatsapp: String(row[2] || ''),
      birthDate: safeFormatStudentDate(row[3]),
      program: String(row[4] || ''),
      level: String(row[5] || ''),
      teacher: String(row[6] || ''),
      supervisor: String(row[7] || ''),
      monthlyLessons: row[8] || '',
      lessonDuration: row[9] || '',
      lessonDays: String(row[10] ?? ''),            // ?? بدل || عشان الصفر
      lessonTime: timeToHHmm_(timeShown[i][0]),
      lessonLink: String(row[12] || ''),
      subscriptionStart: safeFormatStudentDate(row[13]),
      status: String(row[14] || ''),
      notes: String(row[15] || ''),
      subscriptionPrice: Number(row[16]) || 0
    });
  }
  return students;
}

========================= */

function saveStudent_(student) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const studentsSheet =
    ss.getSheetByName('الطلاب');

  const teachersSheet =
    ss.getSheetByName('المدرسون');

  if (!studentsSheet) {
    throw new Error('ورقة الطلاب غير موجودة');
  }

  if (!teachersSheet) {
    throw new Error('ورقة المدرسون غير موجودة');
  }

  if (
    !student ||
    !student.name ||
    String(student.name).trim() === ''
  ) {
    throw new Error('اسم الطالب مطلوب');
  }

  if (
    !student.teacher ||
    String(student.teacher).trim() === ''
  ) {
    throw new Error('المدرس مطلوب');
  }


  /*
   * ==========================================
   * البحث عن المدرس بالاسم
   * ==========================================
   */

  const teachersLastRow =
    teachersSheet.getLastRow();

  if (teachersLastRow < 2) {
    throw new Error('لا يوجد مدرسون مسجلون');
  }

  const teachersData =
    teachersSheet
      .getRange(
        2,
        1,
        teachersLastRow - 1,
        9
      )
      .getValues();

  let selectedTeacher = null;

  const selectedTeacherName =
    String(student.teacher).trim();

  for (
    let i = 0;
    i < teachersData.length;
    i++
  ) {

    const teacherName =
      String(teachersData[i][1] || '').trim();

    if (
      teacherName ===
      selectedTeacherName
    ) {

      selectedTeacher = {

        id: teachersData[i][0],

        name: teachersData[i][1]

      };

      break;
    }
  }


  if (!selectedTeacher) {

    throw new Error(
      'لم يتم العثور على المدرس المختار: ' +
      selectedTeacherName
    );

  }


  /*
   * ==========================================
   * إضافة طالب جديد
   * ==========================================
   */

  let id = student.id;


  if (!id) {

    id = generateStudentId_();


    const row = [

      id,                                  // A رقم الطالب

      student.name || '',                  // B اسم الطالب

      student.whatsapp || '',              // C WhatsApp

      parseDate(student.birthDate),        // D تاريخ الميلاد

      student.program || '',               // E البرنامج

      student.level || '',                 // F المستوى

      selectedTeacher.name || '',          // G المدرس

      student.supervisor || '',            // H المشرف

      student.monthlyLessons || '',        // I عدد الحصص

      student.lessonDuration || '',        // J مدة الحصة

      student.lessonDays || '',            // K أيام الحصص

      student.lessonTime || '',            // L وقت الحصة

      student.lessonLink || '',            // M رابط الحصة

      parseDate(
        student.subscriptionStart
      ),                                   // N بداية الاشتراك

      student.status || 'نشط',             // O الحالة

      student.notes || '',                 // P ملاحظات

      Number(
        student.subscriptionPrice
      ) || 0                               // Q قيمة الاشتراك

    ];


    studentsSheet.appendRow(row);


    /*
     * ==========================================
     * إنشاء جدول الحصص تلقائيًا
     * ==========================================
     */

    const scheduleResult =
      createStudentSchedule_({

        studentId: id,

        studentName: student.name || '',

        teacherId: selectedTeacher.id,

        teacherName: selectedTeacher.name,

        lessonDays:
          student.lessonDays || '',

        subscriptionStart:
          student.subscriptionStart,

        lessonTime:
          student.lessonTime || '',

        lessonDuration:
          student.lessonDuration || 30,

        lessonLink:
          student.lessonLink || ''

      });


    /*
     * ==========================================
     * إنشاء الاشتراك الأول
     * ==========================================
     */

    let subscriptionResult = null;

    try {

      subscriptionResult =
        createInitialSubscription_(id);

    } catch (error) {

      console.log(
        'تعذر إنشاء الاشتراك الأول: ' +
        error.message
      );

    }


    return {

      success: true,

      message:
        'تم إضافة الطالب وإنشاء جدول الحصص بنجاح',

      id: id,

      lessonsCreated:
        scheduleResult
          ? scheduleResult.lessonsCreated
          : 0,

      subscription:
        subscriptionResult

    };

  }


  /*
   * ==========================================
   * تعديل طالب موجود
   * ==========================================
   */

      /*
    * ==========================================
    * تعديل طالب موجود
    * ==========================================
    */

    const lastRow =
      studentsSheet.getLastRow();

    if (lastRow < 2) {
      throw new Error('لم يتم العثور على الطالب');
    }

    const ids =
      studentsSheet
        .getRange(
          2,
          1,
          lastRow - 1,
          1
        )
        .getValues();


    for (
      let i = 0;
      i < ids.length;
      i++
    ) {

      if (
        String(ids[i][0]).trim() !==
        String(id).trim()
      ) {
        continue;
      }

      const rowNumber =
        i + 2;


      /*
      * ==========================================
      * حفظ بيانات الطالب الجديدة
      * ==========================================
      */

      const row = [

        id,

        student.name || '',

        student.whatsapp || '',

        parseDate(student.birthDate),

        student.program || '',

        student.level || '',

        selectedTeacher.name || '',

        student.supervisor || '',

        Number(student.monthlyLessons) || 0,

        Number(student.lessonDuration) || 30,

        student.lessonDays || '',

        student.lessonTime || '',

        student.lessonLink || '',

        parseDate(student.subscriptionStart),

        student.status || 'نشط',

        student.notes || '',

        Number(student.subscriptionPrice) || 0

      ];


      studentsSheet
        .getRange(
          rowNumber,
          1,
          1,
          17
        )
        .setValues([row]);


      /*
      * ==========================================
      * تحديث الاشتراك الحالي
      * ==========================================
      */

      let subscriptionResult = null;

      try {

        subscriptionResult =
          updateCurrentStudentSubscription_({
            studentId: id,
            studentName: student.name || '',
            monthlyLessons:
              Number(student.monthlyLessons) || 0,
            lessonDuration:
              Number(student.lessonDuration) || 30,
            subscriptionPrice:
              Number(student.subscriptionPrice) || 0
          });

      } catch (error) {

        console.log(
          'تعذر تحديث الاشتراك: ' +
          error.message
        );

      }


      /*
      * ==========================================
      * مزامنة جدول الحصص المستقبلية
      *
      * مهم جدًا:
      * لا ننشئ سلسلة جديدة.
      * ==========================================
      */

      let scheduleResult = null;

      try {

        scheduleResult =
          syncStudentFutureLessons_({

            studentId: id,

            studentName:
              student.name || '',

            teacherId:
              selectedTeacher.id,

            teacherName:
              selectedTeacher.name,

            monthlyLessons:
              Number(student.monthlyLessons) || 0,

            lessonDays:
              student.lessonDays || '',

            subscriptionStart:
              student.subscriptionStart,

            lessonTime:
              student.lessonTime || '',

            lessonDuration:
              Number(student.lessonDuration) || 30,

            lessonLink:
              student.lessonLink || ''

          });

      } catch (error) {

        console.log(
          'تعذر مزامنة جدول الحصص: ' +
          error.message
        );

      }


      return {

        success: true,

        message:
          'تم تعديل بيانات الطالب وتحديث الاشتراك والجدول المستقبلي بنجاح',

        id: id,

        lessonsUpdated:
          scheduleResult
            ? scheduleResult.lessonsUpdated
            : 0,

        lessonsCreated:
          scheduleResult
            ? scheduleResult.lessonsCreated
            : 0,

        lessonsDeleted:
          scheduleResult
            ? scheduleResult.lessonsDeleted
            : 0,

        subscription:
          subscriptionResult

      };

    }


    throw new Error(
      'لم يتم العثور على الطالب'
    );
    }

function safeFormatStudentDate(value) {

  if (!value) {
    return '';
  }

  try {

    if (Object.prototype.toString.call(value) === '[object Date]') {

      if (isNaN(value.getTime())) {
        return '';
      }

      return formatDate(value);
    }

    return String(value);

  } catch (error) {

    console.log('خطأ في تنسيق تاريخ الطالب: ' + error);

    return '';

  }
}

/* ======================
   تقارير طالب محدد
====================== */

function getStudentReports_(studentId) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('التقارير');

  if (!sheet) {
    throw new Error('لم يتم العثور على ورقة "التقارير"');
  }

  const data = sheet.getDataRange().getValues();

  if (data.length <= 1) {
    return [];
  }

  const headers = data[0];

  // البحث عن أرقام الأعمدة بالاسم
  const colReportId = headers.indexOf('رقم التقرير');
  const colLessonDate = headers.indexOf('تاريخ الحصة');
  const colStudentId = headers.indexOf('رقم الطالب');
  const colStudentName = headers.indexOf('اسم الطالب');
  const colTeacher = headers.indexOf('المدرس');
  const colSummary = headers.indexOf('ملخص الحصة');
  const colLevel = headers.indexOf('مستوى الطالب');
  const colMemorization = headers.indexOf('مقدار الحفظ');
  const colHomework = headers.indexOf('الواجب');
  const colNotes = headers.indexOf('ملاحظات');
  const colCreatedAt = headers.indexOf('وقت إنشاء التقرير');


  const reports = [];

  for (let i = 1; i < data.length; i++) {

    const row = data[i];

    // تجاهل الصفوف الفارغة
    if (!row || row.length === 0) {
      continue;
    }

    const rowStudentId =
      colStudentId !== -1
        ? String(row[colStudentId]).trim()
        : '';

    // لا نرجع إلا تقارير الطالب المطلوب
    if (rowStudentId !== String(studentId).trim()) {
      continue;
    }


    reports.push({

      id:
        colReportId !== -1
          ? row[colReportId]
          : '',

      date:
        colLessonDate !== -1
          ? formatDateForClient(row[colLessonDate])
          : '',

      studentId:
        colStudentId !== -1
          ? row[colStudentId]
          : '',

      studentName:
        colStudentName !== -1
          ? row[colStudentName]
          : '',

      teacher:
        colTeacher !== -1
          ? row[colTeacher]
          : '',

      summary:
        colSummary !== -1
          ? row[colSummary]
          : '',

      level:
        colLevel !== -1
          ? row[colLevel]
          : '',

      memorization:
        colMemorization !== -1
          ? row[colMemorization]
          : '',

      homework:
        colHomework !== -1
          ? row[colHomework]
          : '',

      notes:
        colNotes !== -1
          ? row[colNotes]
          : '',

      createdAt:
        colCreatedAt !== -1
          ? formatDateForClient(row[colCreatedAt])
          : ''

    });

  }


  // ترتيب التقارير من الأقدم إلى الأحدث
  reports.sort(function(a, b) {

    const dateA = new Date(a.date || 0);
    const dateB = new Date(b.date || 0);

    return dateA - dateB;

  });


  return reports;
}


/* ======================
   تنسيق التاريخ للإرسال
   إلى index.html
====================== */

function formatDateForClient(value) {

  if (!value) {
    return '';
  }

  if (Object.prototype.toString.call(value) === '[object Date]') {

    if (isNaN(value.getTime())) {
      return '';
    }

    return Utilities.formatDate(
      value,
      Session.getScriptTimeZone(),
      'yyyy-MM-dd'
    );

  }

  return String(value);
}

/* =========================
   حفظ طالب
========================= */






function createStudentSchedule_(data) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const lessonsSheet =
    ss.getSheetByName('الحصص');

  if (!lessonsSheet) {

    throw new Error(
      'لم يتم العثور على ورقة الحصص'
    );

  }


  if (!data.studentId) {

    throw new Error(
      'رقم الطالب مطلوب لإنشاء الجدول'
    );

  }


  if (!data.teacherId) {

    throw new Error(
      'رقم المدرس مطلوب لإنشاء الجدول'
    );

  }


  /*
   * ==========================
   * الأيام المختارة
   * ==========================
   */

  const days =
    String(
      data.lessonDays || ''
    )
      .split(',')
      .map(function(day) {
        return Number(day);
      })
      .filter(function(day) {
        return !isNaN(day) &&
               day >= 0 &&
               day <= 6;
      });


  if (days.length === 0) {

    throw new Error(
      'لم يتم تحديد أيام الحصص'
    );

  }


  /*
   * ==========================
   * تاريخ البداية
   * ==========================
   */

  const startDate =
    parseDate(
      data.subscriptionStart
    );


  if (!startDate) {

    throw new Error(
      'تاريخ بداية الاشتراك غير صحيح'
    );

  }


  /*
   * ==========================
   * قراءة الحصص الحالية
   * لمنع التكرار
   * ==========================
   */

  const existingData =
    lessonsSheet
      .getDataRange()
      .getValues();


  const headers =
    existingData[0].map(function(h) {
      return String(h).trim();
    });


  const dateCol =
    headers.indexOf('التاريخ');

  const timeCol =
    headers.indexOf('الوقت');

  const studentIdCol =
    headers.indexOf('رقم الطالب');


  /*
   * مفتاح الحصة:
   *
   * الطالب + التاريخ + الوقت
   */

  const existingLessons = {};


  if (
    dateCol !== -1 &&
    timeCol !== -1 &&
    studentIdCol !== -1
  ) {

    for (
      let i = 1;
      i < existingData.length;
      i++
    ) {

      const row =
        existingData[i];


      const existingStudent =
        String(
          row[studentIdCol] || ''
        ).trim();


      if (
        existingStudent !==
        String(data.studentId).trim()
      ) {

        continue;

      }


      let existingDate =
        row[dateCol];


      let dateKey = '';


      if (
        Object.prototype.toString.call(
          existingDate
        ) === '[object Date]'
      ) {

        dateKey =
          Utilities.formatDate(
            existingDate,
            Session.getScriptTimeZone(),
            'yyyy-MM-dd'
          );

      } else {

        dateKey =
          String(
            existingDate || ''
          ).trim();

      }


      const existingTime =
        String(
          row[timeCol] || ''
        ).trim();


      const key =
        existingStudent +
        '|' +
        dateKey +
        '|' +
        existingTime;


      existingLessons[key] = true;

    }

  }


  /*
   * ==========================
   * تجهيز وقت الحصة
   * ==========================
   */

  const timeParts =
    String(
      data.lessonTime || ''
    ).split(':');


  const hour =
    Number(timeParts[0] || 0);

  const minute =
    Number(timeParts[1] || 0);


  /*
   * ==========================
   * إنشاء الحصص
   * ==========================
   */

  const rowsToAdd = [];


  /*
   * مهم:
   * تخزين أرقام الحصص التي تم توليدها
   * أثناء عملية الإنشاء الحالية.
   *
   * الهدف:
   * ضمان عدم تكرار رقم الحصة
   * حتى لو كانت generateLessonId_()
   * تعطي نفس الرقم أكثر من مرة.
   */

  const generatedLessonIds = {};


  /*
   * 52 أسبوعًا
   */

  for (
    let week = 0;
    week < 52;
    week++
  ) {

    for (
      let i = 0;
      i < days.length;
      i++
    ) {

      const targetDay =
        days[i];


      /*
       * نبدأ من تاريخ الاشتراك
       * ونبحث عن اليوم المختار
       */

      const lessonDate =
        new Date(startDate);


      lessonDate.setHours(
        0,
        0,
        0,
        0
      );


      const currentDay =
        lessonDate.getDay();


      let difference =
        targetDay -
        currentDay;


      if (difference < 0) {

        difference += 7;

      }


      /*
       * إضافة الأسابيع
       */

      difference +=
        week * 7;


      lessonDate.setDate(
        lessonDate.getDate() +
        difference
      );


      lessonDate.setHours(
        hour,
        minute,
        0,
        0
      );


      /*
       * مفتاح منع التكرار
       */

      const dateKey =
        Utilities.formatDate(
          lessonDate,
          Session.getScriptTimeZone(),
          'yyyy-MM-dd'
        );


      const timeKey =
        Utilities.formatDate(
          lessonDate,
          Session.getScriptTimeZone(),
          'HH:mm'
        );


      const key =
        String(data.studentId).trim() +
        '|' +
        dateKey +
        '|' +
        timeKey;


      if (existingLessons[key]) {

        continue;

      }


      /*
       * ==========================
       * توليد رقم حصة فريد
       * ==========================
       */
      let lessonId = createUniqueLessonId_(
          existingData,
          headers,
          generatedLessonIds
        );

      generatedLessonIds[lessonId] = true;
      


      


      /*
       * تجهيز صف الحصة
       */

      const lessonRow =
        new Array(
          headers.length
        ).fill('');


      const setValue =
        function(header, value) {

          const index =
            headers.indexOf(header);

          if (index !== -1) {
            lessonRow[index] =
              value;
          }

        };


      setValue(
        'رقم الحصة',
        lessonId
      );


      setValue(
        'التاريخ',
        lessonDate
      );


      setValue(
        'الوقت',
        timeKey
      );


      setValue(
        'رقم الطالب',
        data.studentId
      );


      setValue(
        'اسم الطالب',
        data.studentName
      );


      setValue(
        'رقم المدرس',
        data.teacherId
      );


      setValue(
        'اسم المدرس',
        data.teacherName
      );


      setValue(
        'مدة الحصة',
        Number(
          data.lessonDuration
        ) || 30
      );


      setValue(
        'رابط الحصة',
        data.lessonLink || ''
      );


      setValue(
        'حالة الحصة',
        'لم تبدأ'
      );


      setValue(
        'وقت الدخول',
        ''
      );


      setValue(
        'وقت الانتهاء',
        ''
      );


      setValue(
        'تقرير الحصة',
        ''
      );


      setValue(
        'أجر المدرس',
        ''
      );


      rowsToAdd.push(
        lessonRow
      );


      /*
       * تسجيل الحصة الجديدة أيضًا
       * لمنع تكرارها داخل نفس العملية.
       */

      existingLessons[key] = true;

    }

  }


  /*
   * ==========================
   * حفظ كل الحصص دفعة واحدة
   * ==========================
   */

  if (rowsToAdd.length > 0) {

    lessonsSheet
      .getRange(
        lessonsSheet.getLastRow() + 1,
        1,
        rowsToAdd.length,
        headers.length
      )
      .setValues(rowsToAdd);

  }


  return {

    success: true,

    lessonsCreated:
      rowsToAdd.length

  };

}

/* =========================================================
   مزامنة الحصص المستقبلية للطالب
========================================================= */

function syncStudentFutureLessons_(data) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const lessonsSheet =
    ss.getSheetByName('الحصص');

  if (!lessonsSheet) {
    throw new Error(
      'لم يتم العثور على ورقة الحصص'
    );
  }


  const studentId =
    String(data.studentId || '').trim();


  if (!studentId) {
    throw new Error(
      'رقم الطالب مطلوب'
    );
  }


  const monthlyLessons =
    Number(data.monthlyLessons) || 0;


  if (monthlyLessons <= 0) {
    throw new Error(
      'عدد الحصص يجب أن يكون أكبر من صفر'
    );
  }


  /*
   * ==========================================
   * الأيام
   * ==========================================
   */

  const days =
    String(data.lessonDays || '')
      .split(',')
      .map(function(day) {
        return Number(day);
      })
      .filter(function(day) {
        return !isNaN(day) &&
               day >= 0 &&
               day <= 6;
      });


  if (days.length === 0) {
    throw new Error(
      'لم يتم تحديد أيام الحصص'
    );
  }


  /*
   * ==========================================
   * تاريخ البداية
   * ==========================================
   */

  const startDate =
    parseDate(
      data.subscriptionStart
    );


  if (!startDate ||
      isNaN(startDate.getTime())) {

    throw new Error(
      'تاريخ بداية الاشتراك غير صحيح'
    );

  }


  /*
   * ==========================================
   * قراءة الحصص
   * ==========================================
   */

  const allData =
    lessonsSheet
      .getDataRange()
      .getValues();


  if (allData.length < 2) {

    /*
     * لا توجد حصص أصلًا
     * فننشئ الجدول المطلوب
     */
    return createFutureLessonsForStudent_(
      data
    );

  }


  const headers =
    allData[0].map(function(h) {
      return String(h).trim();
    });


  const colId =
    headers.indexOf('رقم الحصة');

  const colDate =
    headers.indexOf('التاريخ');

  const colTime =
    headers.indexOf('الوقت');

  const colStudentId =
    headers.indexOf('رقم الطالب');

  const colStatus =
    headers.indexOf('حالة الحصة');

  const colStudentName =
    headers.indexOf('اسم الطالب');

  const colTeacherId =
    headers.indexOf('رقم المدرس');

  const colTeacherName =
    headers.indexOf('اسم المدرس');

  const colDuration =
    headers.indexOf('مدة الحصة');

  const colLink =
    headers.indexOf('رابط الحصة');


  if (
    colStudentId === -1 ||
    colDate === -1 ||
    colStatus === -1
  ) {

    throw new Error(
      'أعمدة الحصص الأساسية غير موجودة'
    );

  }


  /*
   * ==========================================
   * تاريخ ووقت الآن
   * ==========================================
   */

  const now =
    new Date();

  now.setSeconds(0, 0);


  /*
   * ==========================================
   * الحصص التي لا نلمسها
   *
   * مكتملة
   * جارية
   * بانتظار التقرير
   * ==========================================
   */

  const protectedLessons = [];

  /*
   * الحصص المستقبلية التي لم تبدأ
   */
  const futureUnstarted = [];


  /*
   * عدد الحصص المنفذة
   */
  let completedLessons = 0;


  for (
    let i = 1;
    i < allData.length;
    i++
  ) {

    const row =
      allData[i];


    if (
      String(row[colStudentId] || '').trim() !==
      studentId
    ) {
      continue;
    }


    const status =
      String(
        row[colStatus] || ''
      ).trim();


    /*
     * التاريخ
     */
    let lessonDate =
      row[colDate];


    if (
      Object.prototype.toString.call(
        lessonDate
      ) !== '[object Date]'
    ) {

      lessonDate =
        parseDate(lessonDate);

    }


    if (
      !lessonDate ||
      isNaN(lessonDate.getTime())
    ) {
      continue;
    }


    /*
     * نحسب الحصص المكتملة
     */
    if (status === 'مكتملة') {

      completedLessons++;

    }


    /*
     * الحصص المحمية
     */
    if (
      status === 'مكتملة' ||
      status === 'جارية' ||
      status === 'بانتظار التقرير'
    ) {

      protectedLessons.push({

        rowNumber: i + 1,

        row: row,

        date: lessonDate,

        status: status

      });

      continue;

    }


    /*
     * فقط الحصص التي لم تبدأ
     * والتي موعدها اليوم أو بعده
     */
    if (
      status === 'لم تبدأ' &&
      lessonDate >= startOfDay_(now)
    ) {

      futureUnstarted.push({

        rowNumber: i + 1,

        row: row,

        date: lessonDate

      });

    }

  }


  /*
   * ==========================================
   * عدد الحصص التي يجب أن تكون مستقبلية
   *
   * مثال:
   * الاشتراك 8
   * المنفذة 2
   *
   * المطلوب = 6 حصص مستقبلية
   *
   * ونطرح الحصص الجارية/بانتظار التقرير
   * لأنها محفوظة بالفعل.
   * ==========================================
   */

  const protectedFutureCount =
    protectedLessons.filter(function(item) {

      return item.date >= startOfDay_(now) &&
        (
          item.status === 'جارية' ||
          item.status === 'بانتظار التقرير'
        );

    }).length;


  const desiredFutureCount =
    Math.max(
      monthlyLessons -
      completedLessons -
      protectedFutureCount,
      0
    );


  /*
   * ترتيب الحصص المستقبلية
   */
  futureUnstarted.sort(function(a, b) {

    return a.date - b.date;

  });


  /*
   * ==========================================
   * توليد المواعيد المطلوبة
   * ==========================================
   */

  const desiredDates =
    generateNextLessonDates_({

      startDate: startDate,

      days: days,

      time: data.lessonTime,

      count: desiredFutureCount,

      now: now

    });


  /*
   * ==========================================
   * حماية مواعيد الحصص غير القابلة للتعديل
   * ==========================================
   */

  const occupiedKeys = {};


  protectedLessons.forEach(function(item) {

    const dateKey =
      formatLessonDateKey_(item.date);

    const timeKey =
      getLessonTimeKey_(
        item.row[colTime]
      );

    occupiedKeys[
      dateKey + '|' + timeKey
    ] = true;

  });


  /*
   * ==========================================
   * إزالة أي موعد متعارض
   * ==========================================
   */

  const filteredDates = [];

  desiredDates.forEach(function(date) {

    const dateKey =
      formatLessonDateKey_(date);

    const timeKey =
      Utilities.formatDate(
        date,
        Session.getScriptTimeZone(),
        'HH:mm'
      );

    const key =
      dateKey + '|' + timeKey;


    if (occupiedKeys[key]) {
      return;
    }


    occupiedKeys[key] = true;

    filteredDates.push(date);

  });


  /*
   * ==========================================
   * الحصص التي سنحتفظ بها
   * ==========================================
   */

  const keepCount =
    Math.min(
      futureUnstarted.length,
      filteredDates.length
    );


  let lessonsUpdated = 0;
  let lessonsCreated = 0;
  let lessonsDeleted = 0;


  /*
   * ==========================================
   * تحديث الحصص الموجودة
   * ==========================================
   */

  for (
    let i = 0;
    i < keepCount;
    i++
  ) {

    const item =
      futureUnstarted[i];

    const targetDate =
      filteredDates[i];


    updateLessonRowFromStudent_(
      lessonsSheet,
      item.rowNumber,
      headers,
      {
        studentName:
          data.studentName,

        teacherId:
          data.teacherId,

        teacherName:
          data.teacherName,

        duration:
          data.lessonDuration,

        lessonLink:
          data.lessonLink,

        lessonDate:
          targetDate
      }
    );


    lessonsUpdated++;

  }


  /*
   * ==========================================
   * حذف الحصص الزائدة
   *
   * نحذف من آخر المستقبل
   * حتى لا نلمس التاريخ القديم.
   * ==========================================
   */

  if (
    futureUnstarted.length >
    keepCount
  ) {

    for (
      let i = futureUnstarted.length - 1;
      i >= keepCount;
      i--
    ) {

      lessonsSheet.deleteRow(
        futureUnstarted[i].rowNumber
      );

      lessonsDeleted++;

    }

  }


  /*
   * ==========================================
   * إنشاء الحصص الجديدة الناقصة
   * ==========================================
   */

  if (
    filteredDates.length >
    keepCount
  ) {

    const generatedLessonIds = {};

    const currentData =
      lessonsSheet
        .getDataRange()
        .getValues();


    for (
      let i = keepCount;
      i < filteredDates.length;
      i++
    ) {

      const lessonDate =
        filteredDates[i];


      const lessonId =
        createUniqueLessonId_(
          currentData,
          headers,
          generatedLessonIds
        );


      generatedLessonIds[lessonId] = true;


      const lessonRow =
        new Array(
          headers.length
        ).fill('');


      setLessonRowValue_(
        lessonRow,
        headers,
        'رقم الحصة',
        lessonId
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'التاريخ',
        lessonDate
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'الوقت',
        Utilities.formatDate(
          lessonDate,
          Session.getScriptTimeZone(),
          'HH:mm'
        )
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'رقم الطالب',
        studentId
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'اسم الطالب',
        data.studentName
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'رقم المدرس',
        data.teacherId
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'اسم المدرس',
        data.teacherName
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'مدة الحصة',
        Number(data.lessonDuration) || 30
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'رابط الحصة',
        data.lessonLink || ''
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'حالة الحصة',
        'لم تبدأ'
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'وقت الدخول',
        ''
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'وقت الانتهاء',
        ''
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'تقرير الحصة',
        ''
      );


      setLessonRowValue_(
        lessonRow,
        headers,
        'أجر المدرس',
        ''
      );


      lessonsSheet
        .appendRow(
          lessonRow
        );


      lessonsCreated++;

    }

  }


  return {

    success: true,

    lessonsUpdated:
      lessonsUpdated,

    lessonsCreated:
      lessonsCreated,

    lessonsDeleted:
      lessonsDeleted,

    desiredFutureCount:
      desiredFutureCount

  };

}

/* =========================================================
   توليد مواعيد الحصص القادمة
========================================================= */

function generateNextLessonDates_(data) {

  const days =
    Array.isArray(data.days)
      ? data.days
      : [];


  const count =
    Number(data.count) || 0;


  const result = [];


  if (
    days.length === 0 ||
    count <= 0
  ) {
    return result;
  }


  const now =
    new Date(data.now || new Date());

  now.setSeconds(0, 0);


  let cursor =
    new Date(
      data.startDate
    );


  cursor.setHours(
    0,
    0,
    0,
    0
  );


  const today =
    startOfDay_(now);


  if (cursor < today) {
    cursor = today;
  }


  /*
   * حماية من الدوران اللانهائي
   */
  const maxDays =
    370;


  for (
    let offset = 0;
    offset < maxDays &&
    result.length < count;
    offset++
  ) {

    const candidate =
      new Date(cursor);

    candidate.setDate(
      cursor.getDate() + offset
    );


    const day =
      candidate.getDay();


    if (
      days.indexOf(day) === -1
    ) {
      continue;
    }


    const parts =
      String(
        data.time || ''
      ).split(':');


    const hour =
      Number(parts[0]) || 0;

    const minute =
      Number(parts[1]) || 0;


    candidate.setHours(
      hour,
      minute,
      0,
      0
    );


    /*
     * لا نرجع موعدًا مضى
     */
    if (
      candidate < now
    ) {
      continue;
    }


    result.push(
      candidate
    );

  }


  return result;

}

    )
  );


  setLessonRowValue_(
    values,
    headers,

    data.studentName
  );


  setLessonRowValue_(
    values,

    'رقم المدرس',
    data.teacherId
  );


  setLessonRowValue_(
    values,
    headers,
    'اسم المدرس',
    data.teacherName
  );


  setLessonRowValue_(
    values,

  );

/* =========================================================
   وضع قيمة داخل صف الحصة حسب اسم العمود
========================================================= */

function setLessonRowValue_(
  row,
  headers,
  header,
  value
) {

  const index =
    headers.indexOf(header);

  if (index !== -1) {
    row[index] = value;
  }

}


/* =========================================================
   بداية اليوم
========================================================= */

function startOfDay_(date) {

  const d =
    new Date(date);

  d.setHours(
    0,
    0,
    0,
    0
  );

  return d;

}


/* =========================================================
   مفتاح التاريخ
========================================================= */

function formatLessonDateKey_(date) {

  return Utilities.formatDate(
    new Date(date),
    Session.getScriptTimeZone(),
    'yyyy-MM-dd'
  );

}


/* =========================================================
   مفتاح الوقت
========================================================= */

function getLessonTimeKey_(value) {

  if (
    Object.prototype.toString.call(
      value
    ) === '[object Date]'
  ) {

    return Utilities.formatDate(
      value,
      Session.getScriptTimeZone(),
      'HH:mm'
    );

  }


  const text =
    String(value || '').trim();


  const match =
    text.match(
      /(\d{1,2}):(\d{2})/
    );


  if (!match) {
    return text;
  }


  return (
    String(
      Number(match[1])
    ).padStart(2, '0') +
    ':' +
    match[2]
  );

}

/* =========================================================
   مزامنة الحصص القادمة عند تعديل بيانات الطالب
   ========================================================= */

function syncStudentFutureLessons_(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const lessonsSheet = ss.getSheetByName('الحصص');
  if (!lessonsSheet) throw new Error('لم يتم العثور على ورقة الحصص');
  if (!data || !data.studentId) throw new Error('رقم الطالب مطلوب لمزامنة الحصص');
  if (!data.teacherId) throw new Error('رقم المدرس مطلوب لمزامنة الحصص');

  const days = Array.from(new Set(String(data.lessonDays || '').split(',').map(Number).filter(function(d){return !isNaN(d)&&d>=0&&d<=6;}))).sort(function(a,b){return a-b;});
  if (!days.length) throw new Error('لم يتم تحديد أيام الحصص');

  const startDate = parseDate(data.subscriptionStart);
  if (!startDate || isNaN(startDate.getTime())) throw new Error('تاريخ بداية الاشتراك غير صحيح');

  const lastRow = lessonsSheet.getLastRow();
  if (lastRow < 2) return createFutureLessonsForStudent_(data);

  const values = lessonsSheet.getDataRange().getValues();
  const headers = values[0].map(function(h){return String(h).trim();});
  const colDate=headers.indexOf('التاريخ'), colTime=headers.indexOf('الوقت'), colStudentId=headers.indexOf('رقم الطالب'),
        colStudentName=headers.indexOf('اسم الطالب'), colTeacherId=headers.indexOf('رقم المدرس'), colTeacherName=headers.indexOf('اسم المدرس'),
        colDuration=headers.indexOf('مدة الحصة'), colLink=headers.indexOf('رابط الحصة'), colStatus=headers.indexOf('حالة الحصة');
  if(colDate===-1||colStudentId===-1||colStatus===-1) throw new Error('أعمدة ورقة الحصص الأساسية غير موجودة');

  const now=new Date(); now.setSeconds(0,0);
  const today=startOfDay_(now);
  const futureRows=[], occupied={};
  let completed=0, protectedFuture=0;

  for(let i=1;i<values.length;i++){
    const row=values[i];
    if(String(row[colStudentId]||'').trim()!==String(data.studentId).trim()) continue;
    const status=String(row[colStatus]||'').trim();
    let d=row[colDate];
    if(Object.prototype.toString.call(d)!=='[object Date]'||isNaN(d.getTime())) d=parseDate(d);
    if(!d||isNaN(d.getTime())) continue;
    const day=new Date(d); day.setHours(0,0,0,0);

    if(status==='مكتملة') completed++;
    if(status==='جارية'||status==='بانتظار التقرير'||status==='مكتملة'){
      if((status==='جارية'||status==='بانتظار التقرير')&&day>=today) protectedFuture++;
      occupied[Utilities.formatDate(day,Session.getScriptTimeZone(),'yyyy-MM-dd')+'|'+getLessonTimeKey_(row[colTime])]=true;
    } else if(status==='لم تبدأ'&&day>=today){
      futureRows.push({arrayIndex:i,sheetRow:i+1,date:day});
    }
  }

  futureRows.sort(function(a,b){return a.date-b.date;});
  const desiredCount=Math.max((Number(data.monthlyLessons)||0)-completed-protectedFuture,0);

  const parts=String(data.lessonTime||'00:00').split(':'), hour=Number(parts[0]||0), minute=Number(parts[1]||0);
  const cursor=new Date(startDate); cursor.setHours(0,0,0,0);
  const desiredDates=[];

  for(let n=0;n<370&&desiredDates.length<desiredCount;n++){
    if(days.indexOf(cursor.getDay())!==-1){
      const d=new Date(cursor); d.setHours(hour,minute,0,0);
      const key=Utilities.formatDate(d,Session.getScriptTimeZone(),'yyyy-MM-dd')+'|'+Utilities.formatDate(d,Session.getScriptTimeZone(),'HH:mm');
      if(d>=now && !occupied[key]){occupied[key]=true;desiredDates.push(d);}
    }
    cursor.setDate(cursor.getDate()+1);
  }

  const keep=Math.min(futureRows.length,desiredDates.length);
  let updated=0,deleted=0,created=0;

  for(let i=0;i<keep;i++){
    const row=values[futureRows[i].arrayIndex], d=desiredDates[i];
    if(colDate!==-1) row[colDate]=d;
    if(colTime!==-1) row[colTime]=Utilities.formatDate(d,Session.getScriptTimeZone(),'HH:mm');
    if(colStudentName!==-1) row[colStudentName]=data.studentName||'';
    if(colTeacherId!==-1) row[colTeacherId]=data.teacherId||'';
    if(colTeacherName!==-1) row[colTeacherName]=data.teacherName||'';
    if(colDuration!==-1) row[colDuration]=Number(data.lessonDuration)||30;
    if(colLink!==-1) row[colLink]=data.lessonLink||'';
    updated++;
  }

  const rowsToDelete=futureRows.slice(keep).map(function(x){return x.sheetRow;});
  if(rowsToDelete.length){
    let start=rowsToDelete[rowsToDelete.length-1], count=1;
    for(let i=rowsToDelete.length-2;i>=0;i--){
      const r=rowsToDelete[i];
      if(r===start-1){start=r;count++;}
      else{lessonsSheet.deleteRows(start,count);deleted+=count;start=r;count=1;}
    }
    lessonsSheet.deleteRows(start,count); deleted+=count;
  }

  if(updated>0) lessonsSheet.getRange(2,1,values.length-1,headers.length).setValues(values.slice(1));

  if(desiredDates.length>keep){
    const generated={}, rows=[];
    for(let i=keep;i<desiredDates.length;i++){
      const d=desiredDates[i], id=createUniqueLessonId_(values,headers,generated);
      generated[id]=true;
      const r=new Array(headers.length).fill('');
      setLessonRowValue_(r,headers,'رقم الحصة',id);
      setLessonRowValue_(r,headers,'التاريخ',d);
      setLessonRowValue_(r,headers,'الوقت',Utilities.formatDate(d,Session.getScriptTimeZone(),'HH:mm'));
      setLessonRowValue_(r,headers,'رقم الطالب',String(data.studentId).trim());
      setLessonRowValue_(r,headers,'اسم الطالب',data.studentName||'');
      setLessonRowValue_(r,headers,'رقم المدرس',data.teacherId||'');
      setLessonRowValue_(r,headers,'اسم المدرس',data.teacherName||'');
      setLessonRowValue_(r,headers,'مدة الحصة',Number(data.lessonDuration)||30);
      setLessonRowValue_(r,headers,'رابط الحصة',data.lessonLink||'');
      setLessonRowValue_(r,headers,'حالة الحصة','لم تبدأ');
      rows.push(r); created++;
    }
    if(rows.length) lessonsSheet.getRange(lessonsSheet.getLastRow()+1,1,rows.length,headers.length).setValues(rows);
  }

  return {success:true,lessonsUpdated:updated,lessonsCreated:created,lessonsDeleted:deleted,desiredFutureCount:desiredCount};
}

/* =========================
   حذف طالب
========================= */

function deleteStudent_(id) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const studentsSheet = ss.getSheetByName('الطلاب');
  const lessonsSheet = ss.getSheetByName('الحصص');
  if (!studentsSheet) throw new Error('ورقة الطلاب غير موجودة');

  const targetId = String(id).trim();
  let deletedLessons = 0;

  if (lessonsSheet && lessonsSheet.getLastRow() >= 2) {
    const data = lessonsSheet.getDataRange().getValues();
    const headers = data[0].map(function(h){ return String(h).trim(); });
    const studentIdCol = headers.indexOf('رقم الطالب');
    const dateCol = headers.indexOf('التاريخ');
    const statusCol = headers.indexOf('حالة الحصة');

    if (studentIdCol !== -1 && dateCol !== -1 && statusCol !== -1) {
      const today = startOfDay_(new Date());
      const rowsToDelete = [];

      for (let i=1;i<data.length;i++) {
        const row=data[i];
        if (String(row[studentIdCol]||'').trim() !== targetId) continue;
        if (String(row[statusCol]||'').trim() !== 'لم تبدأ') continue;

        let d=row[dateCol];
        if (Object.prototype.toString.call(d)!=='[object Date]') d=parseDate(d);
        if (!d || isNaN(d.getTime())) continue;
        if (d >= today) rowsToDelete.push(i+1);
      }

      if (rowsToDelete.length) {
        let start=rowsToDelete[rowsToDelete.length-1], count=1;
        for(let i=rowsToDelete.length-2;i>=0;i--){
          const r=rowsToDelete[i];
          if(r===start-1){ start=r; count++; }
          else {
            lessonsSheet.deleteRows(start,count);
            deletedLessons += count;
            start=r; count=1;
          }
        }
        lessonsSheet.deleteRows(start,count);
        deletedLessons += count;
      }
    }
  }

  const data = studentsSheet.getDataRange().getValues();
  for(let i=1;i<data.length;i++){
    if(String(data[i][0]).trim() !== targetId) continue;
    studentsSheet.deleteRow(i+1);
    return {success:true,message:'تم حذف الطالب وحذف الحصص المستقبلية فقط',deletedLessons:deletedLessons};
  }
  throw new Error('لم يتم العثور على الطالب');
}

========================= */

function formatDate(value) {

  if (!value) return '';

  if (Object.prototype.toString.call(value) === '[object Date]') {

    return Utilities.formatDate(
      value,
      Session.getScriptTimeZone(),
      'yyyy-MM-dd'
    );
  }

  return value;
}


function parseDate(value) {

  if (!value) return '';

  if (value instanceof Date) {
    return value;
  }

  return new Date(value);
}

function getStudentProfile_(studentId) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const studentsSheet = ss.getSheetByName('الطلاب');

  if (!studentsSheet) {
    throw new Error('لم يتم العثور على ورقة الطلاب');
  }

  if (!studentId) {
    return null;
  }

  const targetId = String(studentId).trim();

  const lastRow = studentsSheet.getLastRow();

  if (lastRow <= 1) {
    return null;
  }

  // ورقة الطلاب أصبحت 17 عمودًا
  const data = studentsSheet
    .getRange(2, 1, lastRow - 1, 17)
    .getValues();

  let studentRow = null;
  let studentIndex = -1;

  // البحث عن الطالب
  for (let i = 0; i < data.length; i++) {

    const currentId =
      String(data[i][0] ?? '').trim();

    if (currentId === targetId) {

      studentRow = data[i];
      studentIndex = i;
      break;
    }
  }

  if (!studentRow) {
    return null;
  }

  // ==============================
  // بيانات الطالب
  // ==============================

  const student = {

    id:
      String(studentRow[0] ?? ''),

    name:
      String(studentRow[1] ?? ''),

    whatsapp:
      String(studentRow[2] ?? ''),

    birthDate:
      safeFormatStudentDate(studentRow[3]),

    program:
      String(studentRow[4] ?? ''),

    level:
      String(studentRow[5] ?? ''),

    teacher:
      String(studentRow[6] ?? ''),

    supervisor:
      String(studentRow[7] ?? ''),

    monthlyLessons:
      studentRow[8] ?? '',

    lessonDuration:
      studentRow[9] ?? '',

    lessonDays:
      String(studentRow[10] ?? ''),

    lessonTime:
      timeToHHmm_(
        studentsSheet.getRange(studentIndex + 2, 12).getDisplayValue()
      ),

    lessonLink:
      String(studentRow[12] ?? ''),

    subscriptionStart:
      safeFormatStudentDate(studentRow[13]),

    status:
      String(studentRow[14] ?? ''),

    notes:
      String(studentRow[15] ?? ''),

    subscription: null
  };


  // ==============================
  // جلب أحدث اشتراك للطالب
  // ==============================

  const subscriptionsSheet =
    ss.getSheetByName('الاشتراكات');

  if (subscriptionsSheet) {

    const subscriptionsLastRow =
      subscriptionsSheet.getLastRow();

    if (subscriptionsLastRow > 1) {

      const subscriptionsData =
        subscriptionsSheet
          .getRange(
            2,
            1,
            subscriptionsLastRow - 1,
            11
          )
          .getValues();

      let subscription = null;

      // نبحث من آخر اشتراك إلى أول اشتراك
      for (
        let i = subscriptionsData.length - 1;
        i >= 0;
        i--
      ) {

        const subscriptionStudentId =
          String(
            subscriptionsData[i][1] ?? ''
          ).trim();

        if (
          subscriptionStudentId === targetId
        ) {

          subscription = {

            id:
              subscriptionsData[i][0] ?? '',

            studentId:
              subscriptionsData[i][1] ?? '',

            studentName:
              subscriptionsData[i][2] ?? '',

            lessons:
              Number(
                subscriptionsData[i][3]
              ) || 0,

            duration:
              Number(
                subscriptionsData[i][4]
              ) || 0,

            hourlyRate:
              Number(
                subscriptionsData[i][5]
              ) || 0,

            price:
              Number(
                subscriptionsData[i][6]
              ) || 0,

            paymentDate:
              safeFormatStudentDate(
                subscriptionsData[i][7]
              ),

            completed:
              Number(
                subscriptionsData[i][8]
              ) || 0,

            remaining:
              Number(
                subscriptionsData[i][9]
              ) || 0,

            status:
              String(
                subscriptionsData[i][10] ?? ''
              )

          };

          break;
        }
      }

      student.subscription =
        subscription;
    }
  }

  return student;
}

function calculateTeacherPay_(teacherId, duration) {

  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName('المدرسون');

  if (!sheet) {
    throw new Error('لم يتم العثور على ورقة المدرسين');
  }

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return 0;
  }

  const data = sheet
    .getRange(2, 1, lastRow - 1, 9)
    .getValues();

  let hourlyRate = 0;

  for (let i = 0; i < data.length; i++) {

    if (String(data[i][0]) === String(teacherId)) {

      hourlyRate = Number(data[i][4]) || 0;

      break;
    }
  }

  if (hourlyRate <= 0) {
    throw new Error('لم يتم العثور على أجر الساعة للمدرس');
  }

  duration = Number(duration) || 30;

  return hourlyRate * (duration / 60);
}

/**
 * حفظ تقرير الحصة وإنهاء الحصة وحساب أجر المدرس
 * كل ذلك في عملية واحدة
 */
function saveLessonReport_(lessonId, report) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const lessonsSheet = ss.getSheetByName('الحصص');
  const reportsSheet = ss.getSheetByName('التقارير');

  if (!lessonsSheet) {
    throw new Error('لم يتم العثور على ورقة الحصص');
  }

  if (!reportsSheet) {
    throw new Error('لم يتم العثور على ورقة التقارير');
  }

  const lessonsLastRow = lessonsSheet.getLastRow();

  if (lessonsLastRow < 2) {
    throw new Error('لا توجد حصص مسجلة');
  }

  const lessonsData = lessonsSheet
    .getRange(2, 1, lessonsLastRow - 1, 14)
    .getValues();

  let lessonRow = -1;
  let lesson = null;

  for (let i = 0; i < lessonsData.length; i++) {

    if (String(lessonsData[i][0]) === String(lessonId)) {

      lessonRow = i + 2;

      lesson = {
        id: lessonsData[i][0],
        date: lessonsData[i][1],
        time: lessonsData[i][2],
        studentId: lessonsData[i][3],
        studentName: lessonsData[i][4],
        teacherId: lessonsData[i][5],
        teacherName: lessonsData[i][6],
        duration: Number(lessonsData[i][7]) || 30,
        lessonLink: lessonsData[i][8],
        status: lessonsData[i][9],
        startTime: lessonsData[i][10],
        endTime: lessonsData[i][11],
        reportId: lessonsData[i][12],
        teacherPay: lessonsData[i][13]
      };

      break;
    }
  }

  if (lessonRow === -1 || !lesson) {
    throw new Error('لم يتم العثور على الحصة');
  }

  if (lesson.status === 'مكتملة') {
    throw new Error('تم تسجيل تقرير هذه الحصة من قبل');
  }

  /*
   * إنشاء رقم التقرير
   */
  const reportId = generateReportId_();

  const createdAt = new Date();

  /*
   * حساب أجر المدرس
   */
  const teacherPay = calculateTeacherPay_(
    lesson.teacherId,
    lesson.duration
  );

  /*
   * إضافة التقرير إلى ورقة التقارير
   *
   * الترتيب:
   * 1 رقم التقرير
   * 2 تاريخ الحصة
   * 3 رقم الطالب
   * 4 اسم الطالب
   * 5 المدرس
   * 6 ملخص الحصة
   * 7 مستوى الطالب
   * 8 مقدار الحفظ

  reportsSheet.appendRow([
    reportId,
    lesson.date || new Date(),
    lesson.studentId,
    lesson.studentName,
    lesson.teacherName,
    report.summary || '',
    report.level || '',
    report.memorization || '',

  ]);

    reportId: reportId,
    teacherPay: teacherPay
  };
}

function formatTime(value) {

  if (!value) return '';

  if (Object.prototype.toString.call(value) === '[object Date]') {

    return Utilities.formatDate(
      value,
      Session.getScriptTimeZone(),
      'h:mm a'
    );

  }

  // لو كانت القيمة نصًا
  const text = String(value).trim();

  // محاولة قراءة وقت بصيغة HH:mm
  const match = text.match(/(\d{1,2}):(\d{2})/);

  if (match) {

    let hour = parseInt(match[1], 10);
    const minute = match[2];

    const suffix = hour >= 12 ? 'PM' : 'AM';

    hour = hour % 12;

    if (hour === 0) {
      hour = 12;
    }

    return hour + ':' + minute + ' ' + suffix;
  }

  return text;
}

/* =========================
   الاشتراكات
========================= */

function generateSubscriptionId_() {

  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName('الاشتراكات');

  if (!sheet) {
    throw new Error('ورقة الاشتراكات غير موجودة');
  }

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return 'SUB-0001';
  }

  const ids = sheet
    .getRange(2, 1, lastRow - 1, 1)
    .getValues();

  let maxNumber = 0;

  ids.forEach(function(row) {

    const id = String(row[0] || '');

    const match = id.match(/^SUB-(\d+)$/);

    if (match) {

      const number =
        parseInt(match[1], 10);

      if (number > maxNumber) {
        maxNumber = number;
      }

    }

  });

  return 'SUB-' +
    String(maxNumber + 1).padStart(4, '0');
}

function createInitialSubscription_(studentId) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const studentsSheet =
    ss.getSheetByName('الطلاب');

  const subscriptionsSheet =
    ss.getSheetByName('الاشتراكات');

  if (!studentsSheet) {
    throw new Error('ورقة الطلاب غير موجودة');
  }

  if (!subscriptionsSheet) {
    throw new Error('ورقة الاشتراكات غير موجودة');
  }

  /*
   * =========================
   * البحث عن الطالب
   * =========================
   */

  const studentData =
    studentsSheet.getDataRange().getValues();

  let student = null;

  for (let i = 1; i < studentData.length; i++) {

    if (
      String(studentData[i][0]).trim() ===
      String(studentId).trim()
    ) {

      student = {

        id:
          studentData[i][0],

        name:
          studentData[i][1],

        // العمود I = عدد الحصص
        monthlyLessons:
          studentData[i][8],

        // العمود J = مدة الحصة
        lessonDuration:
          studentData[i][9],

        // العمود N = تاريخ بداية الاشتراك
        subscriptionStart:
          studentData[i][13],

        // العمود Q = قيمة الاشتراك
        subscriptionPrice:
          studentData[i][16]

      };

      break;
    }
  }

  if (!student) {

    throw new Error(
      'لم يتم العثور على الطالب لإنشاء الاشتراك'
    );

  }

  const lessons =
    Number(student.monthlyLessons) || 0;

  const duration =
    Number(student.lessonDuration) || 0;

  const subscriptionPrice =
    Number(student.subscriptionPrice) || 0;


  if (lessons <= 0) {

    throw new Error(
      'عدد حصص الاشتراك للطالب غير صحيح'
    );

  }


  if (subscriptionPrice <= 0) {

    throw new Error(
      'قيمة الاشتراك للطالب غير محددة'
    );

  }


  /*
   * =========================
   * التأكد من عدم وجود
   * دورة حالية لها حصص متبقية
   * =========================
   */

  const lastRow =
    subscriptionsSheet.getLastRow();

  if (lastRow >= 2) {

    const existingData =
      subscriptionsSheet
        .getRange(
          2,
          1,
          lastRow - 1,
          11
        )
        .getValues();

    for (
      let i = 0;
      i < existingData.length;
      i++
    ) {

      const row =
        existingData[i];

      const existingStudentId =
        String(row[1] || '').trim();

      const remainingLessons =
        Number(row[9]) || 0;

      if (
        existingStudentId ===
          String(studentId).trim() &&
        remainingLessons > 0
      ) {

        return {

          success: true,

          message:
            'الطالب لديه دورة اشتراك حالية بالفعل',

          id: row[0]

        };

      }

    }

  }


  /*
   * =========================
   * إنشاء رقم الاشتراك
   * =========================
   */

  const subscriptionId =
    generateSubscriptionId_();


  /*
   * =========================
   * إنشاء دورة الاشتراك
   *
   * الدورة الجديدة غير مدفوعة
   * لذلك حالتها = منتهي
   *
   * ولا نضع تاريخ دفع
   * إلا عند الضغط على "دفع"
   * =========================
   */

  subscriptionsSheet.appendRow([

    subscriptionId,          // 1 رقم الاشتراك

    student.id,              // 2 رقم الطالب

    student.name,            // 3 اسم الطالب

    lessons,                 // 4 عدد الحصص

    duration,                // 5 مدة الحصة

    0,                       // 6 سعر الساعة

    subscriptionPrice,       // 7 سعر الاشتراك

    '',                      // 8 تاريخ الدفع

    0,                       // 9 الحصص المنفذة

    lessons,                 // 10 الحصص المتبقية

    'منتهي'                  // 11 الحالة

  ]);


  return {

    success: true,

    id: subscriptionId,

    studentId: student.id,

    totalLessons: lessons,

    completedLessons: 0,

    remainingLessons: lessons,

    price: subscriptionPrice,

    status: 'منتهي'

  };

}

/* =========================================================
   تحديث الاشتراك الحالي عند تعديل بيانات الطالب

function updateCurrentStudentSubscription_(data) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const sheet =
    ss.getSheetByName('الاشتراكات');

  if (!sheet) {
    throw new Error(
      'ورقة الاشتراكات غير موجودة'
    );
  }


  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {

    return {
      success: false,
      message: 'لا يوجد اشتراك حالي للطالب'
    };

  }


  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        11
      )
      .getValues();


  let subscriptionIndex = -1;


  /*
   * نأخذ أحدث اشتراك للطالب
   */
  for (
    let i = rows.length - 1;
    i >= 0;
    i--
  ) {

    if (
      String(rows[i][1] || '').trim() ===
      String(data.studentId).trim()
    ) {

      subscriptionIndex = i;

      break;

    }

  }


  /*
   * لا يوجد اشتراك
   */
  if (subscriptionIndex === -1) {

    return {
      success: false,
      message: 'لم يتم العثور على اشتراك حالي للطالب'
    };

  }


  const rowNumber =
    subscriptionIndex + 2;

  const oldRow =
    rows[subscriptionIndex];


  /*
   * الحصص التي تم تنفيذها بالفعل
   */
  const completedLessons =
    Number(oldRow[8]) || 0;


  /*
   * العدد الجديد
   */
  const newTotalLessons =
    Number(data.monthlyLessons) || 0;


  /*
   * المتبقي الجديد
   */
  const newRemainingLessons =
    Math.max(
      newTotalLessons - completedLessons,
      0
    );


  /*
   * الاحتفاظ بحالة الدفع الحالية
   */
  const oldStatus =
    String(oldRow[10] || '').trim();

  const isPaid =
    oldStatus === 'تم الدفع' ||
    !!oldRow[7];


  const newStatus =
    isPaid
      ? 'تم الدفع'
      : 'منتهي';


  /*
   * تحديث اسم الطالب
   */
  sheet
    .getRange(rowNumber, 3)
    .setValue(
      data.studentName || ''
    );


  /*
   * تحديث عدد الحصص
   */
  sheet
    .getRange(rowNumber, 4)
    .setValue(
      newTotalLessons
    );


  /*
   * تحديث مدة الحصة
   */
  sheet
    .getRange(rowNumber, 5)
    .setValue(
      Number(data.lessonDuration) || 30
    );


  /*
   * تحديث سعر الاشتراك
   */
  sheet
    .getRange(rowNumber, 7)
    .setValue(
      Number(data.subscriptionPrice) || 0
    );


  /*
   * تحديث المتبقي
   */
  sheet
    .getRange(rowNumber, 10)
    .setValue(
      newRemainingLessons
    );


  /*
   * تحديث الحالة
   */
  sheet
    .getRange(rowNumber, 11)
    .setValue(
      newStatus
    );


  return {

    success: true,

    id:
      oldRow[0],

    totalLessons:
      newTotalLessons,

    completedLessons:
      completedLessons,

    remainingLessons:
      newRemainingLessons,

    price:
      Number(data.subscriptionPrice) || 0,

    status:
      newStatus

  };

}

function completeLessonSubscription_(studentId) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const sheet =
    ss.getSheetByName('الاشتراكات');

  if (!sheet) {
    throw new Error(
      'ورقة الاشتراكات غير موجودة'
    );
  }

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    throw new Error(
      'لا توجد اشتراكات'
    );
  }

  const data =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        11
      )
      .getValues();

  /*
   * نبحث من أسفل لأعلى
   * حتى نأخذ أحدث اشتراك للطالب.
   *
   * لا نهتم بالحالة هنا.
   *
   * سواء كانت:
   * منتهي
   * تم الدفع
   *
   * المهم أنه أحدث اشتراك.
   */
  let subscriptionIndex = -1;

  for (let i = data.length - 1; i >= 0; i--) {

    if (
      String(data[i][1] || '').trim() ===
      String(studentId).trim()
    ) {

      subscriptionIndex = i;
      break;

    }

  }

  /*
   * إذا لم يوجد أي اشتراك
   */
  if (subscriptionIndex === -1) {

    throw new Error(
      'لم يتم العثور على اشتراك للطالب'
    );

  }

  const row =
    data[subscriptionIndex];

  const rowNumber =
    subscriptionIndex + 2;

  const totalLessons =
    Number(row[3]) || 0;

  let completed =
    Number(row[8]) || 0;

  let remaining =
    Number(row[9]) || 0;

  /*
   * حماية من القيم الخاطئة
   */
  if (remaining <= 0) {

    /*
     * لو الاشتراك الحالي انتهى بالفعل،
     * ننشئ دورة جديدة تلقائيًا.
     */
    const newSubscription =
      createNextSubscriptionCycle_(
        studentId
      );

    /*
     * نبدأ الحصة الحالية على الاشتراك الجديد
     */
    return completeLessonSubscription_(
      studentId
    );
  }

  /*
   * تنفيذ الحصة
   */
  completed++;

  remaining =
    Math.max(
      totalLessons - completed,
      0
    );

  /*
   * إذا انتهت هذه الدورة
   */
  const wasPaid =
    String(row[10] || '').trim() === 'تم الدفع' || !!row[7];

  const status = wasPaid ? 'تم الدفع' : 'منتهي';
  /*
   * تحديث الحصص المنفذة
   */
  sheet
    .getRange(rowNumber, 9)
    .setValue(completed);

  /*
   * تحديث الحصص المتبقية
   */
  sheet
    .getRange(rowNumber, 10)
    .setValue(remaining);

  /*
   * تحديث الحالة
   */
  sheet
    .getRange(rowNumber, 11)
    .setValue(status);

  /*
   * إذا انتهت الدورة الحالية الآن
   * ننشئ الدورة التالية تلقائيًا.
   */
  let nextSubscription = null;

  if (remaining === 0) {

    nextSubscription =
      createNextSubscriptionCycle_(
        studentId
      );

  }

  return {

    success: true,

    subscriptionId:
      row[0],

    completed:
      completed,

    remaining:
      remaining,

    status:
      status,

    nextSubscription:
      nextSubscription

  };

}

function createNextSubscriptionCycle_(studentId) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const studentsSheet =
    ss.getSheetByName('الطلاب');

  const subscriptionsSheet =
    ss.getSheetByName('الاشتراكات');

  if (!studentsSheet) {
    throw new Error(
      'ورقة الطلاب غير موجودة'
    );
  }

  if (!subscriptionsSheet) {
    throw new Error(
      'ورقة الاشتراكات غير موجودة'
    );
  }

  /*
   * البحث عن الطالب
   */
  const studentsData =
    studentsSheet
      .getDataRange()
      .getValues();

  let student = null;

  for (
    let i = 1;
    i < studentsData.length;
    i++
  ) {

    if (
      String(studentsData[i][0] || '').trim() ===
      String(studentId).trim()
    ) {

      student = {

        id:
          studentsData[i][0],

        name:
          studentsData[i][1],

        lessons:
          Number(studentsData[i][8]) || 0,

        duration:
          Number(studentsData[i][9]) || 0,

        price:
          Number(studentsData[i][16]) || 0

      };

      break;

    }

  }

  if (!student) {

    throw new Error(
      'لم يتم العثور على الطالب لإنشاء الاشتراك الجديد'
    );

  }

  if (student.lessons <= 0) {

    throw new Error(
      'عدد حصص الاشتراك للطالب غير صحيح'
    );

  }

  if (student.price <= 0) {

    throw new Error(
      'قيمة الاشتراك للطالب غير محددة'
    );

  }

  /*
   * حماية مهمة:
   *
   * لا ننشئ اشتراكًا جديدًا إذا كان
   * هناك اشتراك جديد بالفعل لم يبدأ بعد.
   *
   * نتحقق من آخر اشتراك للطالب.
   */
  const lastRow =
    subscriptionsSheet.getLastRow();

  if (lastRow >= 2) {

    const data =
      subscriptionsSheet
        .getRange(
          2,
          1,
          lastRow - 1,
          11
        )
        .getValues();

    for (
      let i = data.length - 1;
      i >= 0;
      i--
    ) {

      if (
        String(data[i][1] || '').trim() ===
        String(studentId).trim()
      ) {

        const latestRemaining =
          Number(data[i][9]) || 0;

        /*
         * لو آخر اشتراك ما زال فيه حصص،
         * فلا ننشئ اشتراكًا آخر.
         */
        if (latestRemaining > 0) {

          return {

            success: true,

            id:
              data[i][0],

            message:
              'يوجد اشتراك حالي للطالب بالفعل'

          };

        }

        /*
         * آخر اشتراك منتهي
         * وبالتالي نكمل لإنشاء الجديد.
         */
        break;

      }

    }

  }

  /*
   * إنشاء رقم اشتراك جديد
   */
  const subscriptionId =
    generateSubscriptionId_();

 

  /*
   * إنشاء الاشتراك الجديد
   *
   * الحالة تبدأ بـ "منتهي"
   * حسب النظام الذي طلبته:
   *
   * منتهي = لم يتم الدفع بعد
   *
   * وعند الضغط على دفع:
   *
   * تم الدفع
   */
  subscriptionsSheet.appendRow([
    subscriptionId,    // 1 رقم الاشتراك
    student.id,        // 2 رقم الطالب
    student.name,      // 3 اسم الطالب
    student.lessons,   // 4 عدد الحصص
    student.duration,  // 5 مدة الحصة
    0,                 // 6 سعر الساعة
    student.price,     // 7 سعر الاشتراك
    '',                // 8 تاريخ الدفع (فاضي لحد ما يدفع)
    0,                 // 9 الحصص المنفذة
    student.lessons,   // 10 الحصص المتبقية
    'منتهي'            // 11 الحالة
  ]);

  return {

    success: true,

    id:
      subscriptionId,

    studentId:
      student.id,

    studentName:
      student.name,

    totalLessons:
      student.lessons,

    completedLessons:
      0,

    remainingLessons:
      student.lessons,

    price:
      student.price,

    status:
      'منتهي'

  };

}

function repairMissingSubscriptions() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const students = ss.getSheetByName('الطلاب').getDataRange().getValues();
  const subs = ss.getSheetByName('الاشتراكات').getDataRange().getValues();

  const latestRemaining = {};
  for (let i = 1; i < subs.length; i++) {
    latestRemaining[String(subs[i][1]).trim()] = Number(subs[i][9]) || 0;
  }

  const fixed = [];
  for (let i = 1; i < students.length; 
       i++) {
    const id = String(students[i][0] || '').trim();
    if (!id) continue;
    if (latestRemaining.hasOwnProperty(id) && latestRemaining[id] === 0) {
      createNextSubscriptionCycle_(id);
      fixed.push(id);
    }
  }
  Logger.log('تم إنشاء دورات جديدة للطلاب: ' + fixed.join(', '));
}

  });

function markSubscriptionAsPaid_(subscriptionId) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName('الاشتراكات');

  if (!sheet) throw new Error('ورقة الاشتراكات غير موجودة');

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) throw new Error('لا توجد اشتراكات');

  const data = sheet.getRange(2, 1, lastRow - 1, 11).getValues();

  for (let i = 0; i < data.length; i++) {
    if (String(data[i][0]).trim() !== String(subscriptionId).trim()) continue;

    const alreadyPaid =
      String(data[i][10] || '').trim() === 'تم الدفع' || !!data[i][7];

    if (alreadyPaid) {
      throw new Error('هذا الاشتراك مدفوع بالفعل');
    }

    const rowNumber = i + 2;
    sheet.getRange(rowNumber, 11).setValue('تم الدفع');
    sheet.getRange(rowNumber, 8).setValue(new Date());

    return {
      success: true,
      id: data[i][0],
      studentId: data[i][1],
      status: 'تم الدفع',
      message: 'تم تسجيل دفع الاشتراك بنجاح'
    };
  }

  throw new Error('لم يتم العثور على الاشتراك');
}

/**
 * جلب رواتب المدرسين لشهر محدد
 *
 * month = yyyy-MM
 * مثال:
 * 2026-09
 */

function getPayroll_(month) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const teachersSheet =
    ss.getSheetByName('المدرسون');

  const lessonsSheet =
    ss.getSheetByName('الحصص');

  const payrollSheet =
    ss.getSheetByName('الرواتب');


  if (!teachersSheet) {
    throw new Error(
      'لم يتم العثور على ورقة "المدرسون".'
    );
  }


  if (!lessonsSheet) {
    throw new Error(
      'لم يتم العثور على ورقة "الحصص".'
    );
  }


  if (!payrollSheet) {
    throw new Error(
      'لم يتم العثور على ورقة "الرواتب".'
    );
  }


  /* ==========================
     التأكد من الشهر
  ========================== */

  if (!month) {

    const now = new Date();

    month =
      Utilities.formatDate(
        now,
        Session.getScriptTimeZone(),
        'yyyy-MM'
      );

  }


  /*
   * توحيد شكل الشهر
   *
   * مثال:
   * 2026-09
   * 2026/09
   * 2026-09-01
   *
   * كلها تتحول إلى:
   * 2026-09
   */

  function normalizeMonth(value) {

    if (!value) {
      return '';
    }


    /*
     * إذا كانت القيمة Date
     */

    if (
      Object.prototype.toString.call(value)
      === '[object Date]'
    ) {

      if (isNaN(value.getTime())) {
        return '';
      }


      return Utilities.formatDate(
        value,
        Session.getScriptTimeZone(),
        'yyyy-MM'
      );

    }


    let text =
      String(value).trim();


    if (!text) {
      return '';
    }


    /*
     * تحويل / إلى -
     */

    text =
      text.replace(/\//g, '-');


    /*
     * إذا كانت القيمة مثل:
     * 2026-09-01
     */

    const match =
      text.match(
        /^(\d{4})-(\d{1,2})/
      );


    if (match) {

      return (
        match[1] +
        '-' +
        String(
          match[2]
        ).padStart(2, '0')
      );

    }


    return text;

  }


  const targetMonth =
    normalizeMonth(month);


  /* ==========================
     قراءة المدرسين
  ========================== */

  const teachersData =
    teachersSheet
      .getDataRange()
      .getValues();


  if (teachersData.length < 2) {
    return [];
  }


  const teacherHeaders =
    teachersData[0].map(
      function(h) {
        return String(h).trim();
      }
    );


  const teacherIdCol =
    teacherHeaders.indexOf(
      'رقم المدرس'
    );


  const teacherNameCol =
    teacherHeaders.indexOf(
      'اسم المدرس'
    );


  const hourlyRateCol =
    teacherHeaders.indexOf(
      'أجر الساعة'
    );


  if (
    teacherIdCol === -1 ||
    teacherNameCol === -1 ||
    hourlyRateCol === -1
  ) {

    throw new Error(
      'تأكد من وجود الأعمدة: رقم المدرس، اسم المدرس، أجر الساعة في ورقة المدرسين.'
    );

  }


  /* ==========================
     قراءة الحصص
  ========================== */

  const lessonsData =
    lessonsSheet
      .getDataRange()
      .getValues();


  const lessonHeaders =
    lessonsData.length
      ? lessonsData[0].map(
          function(h) {
            return String(h).trim();
          }
        )
      : [];


  const lessonDateCol =
    lessonHeaders.indexOf(
      'التاريخ'
    );


  const lessonTeacherIdCol =
    lessonHeaders.indexOf(
      'رقم المدرس'
    );


  const lessonDurationCol =
    lessonHeaders.indexOf(
      'مدة الحصة'
    );


  const lessonStatusCol =
    lessonHeaders.indexOf(
      'حالة الحصة'
    );


  const lessonReportCol =
    lessonHeaders.indexOf(
      'تقرير الحصة'
    );


  if (
    lessonDateCol === -1 ||
    lessonTeacherIdCol === -1 ||
    lessonDurationCol === -1
  ) {

    throw new Error(
      'تأكد من وجود أعمدة التاريخ، رقم المدرس، مدة الحصة في ورقة الحصص.'
    );

  }


  /* ==========================
     قراءة بيانات الرواتب
  ========================== */

  const payrollData =
    payrollSheet
      .getDataRange()
      .getValues();


  const payrollHeaders =
    payrollData.length
      ? payrollData[0].map(
          function(h) {
            return String(h).trim();
          }
        )
      : [];


  const payrollTransactionCol =
    payrollHeaders.indexOf(
      'رقم العملية'
    );


  const payrollMonthCol =
    payrollHeaders.indexOf(
      'الشهر'
    );


  const payrollTeacherIdCol =
    payrollHeaders.indexOf(
      'رقم المدرس'
    );


  const payrollTeacherNameCol =
    payrollHeaders.indexOf(
      'اسم المدرس'
    );


  const payrollLessonsCol =
    payrollHeaders.indexOf(
      'عدد الحصص'
    );


  const payrollHoursCol =
    payrollHeaders.indexOf(
      'إجمالي الساعات'
    );


  const payrollRateCol =
    payrollHeaders.indexOf(
      'أجر الساعة'
    );


  const payrollBonusCol =
    payrollHeaders.indexOf(
      'الحوافز'
    );


  const payrollDeductionCol =
    payrollHeaders.indexOf(
      'الخصومات'
    );


  const payrollTotalCol =
    payrollHeaders.indexOf(
      'إجمالي المستحق'
    );


  /*
   * ملاحظة:
   * عمود الحالة عندك اسمه:
   * الحالة ( تم الدفع/ لم يتم)
   */

  let payrollStatusCol =
    payrollHeaders.indexOf(
      'الحالة ( تم الدفع/ لم يتم)'
    );


  /*
   * احتياطًا لو تغير اسم العمود
   */

  if (payrollStatusCol === -1) {

    payrollStatusCol =
      payrollHeaders.findIndex(
        function(header) {

          return String(header)
            .trim()
            .startsWith('الحالة');

        }
      );

  }


  if (
    payrollMonthCol === -1 ||
    payrollTeacherIdCol === -1
  ) {

    throw new Error(
      'تأكد من وجود عمودي الشهر ورقم المدرس في ورقة الرواتب.'
    );

  }


  /* ==========================
     إنشاء خريطة بيانات الرواتب
  ========================== */

  const payrollMap = {};


  for (
    let i = 1;
    i < payrollData.length;
    i++
  ) {

    const row =
      payrollData[i];


    const rowMonth =
      normalizeMonth(
        row[payrollMonthCol]
      );


    const rowTeacherId =
      String(
        row[payrollTeacherIdCol] || ''
      ).trim();


    if (
      !rowMonth ||
      !rowTeacherId
    ) {

      continue;

    }


    /*
     * نستخدم الشهر بعد توحيده
     */

    if (
      rowMonth !== targetMonth
    ) {

      continue;

    }


    /*
     * مهم جدًا:
     *
     * لو كان هناك أكثر من سجل لنفس المدرس
     * في نفس الشهر، نأخذ آخر سجل.
     *
     * لأن آخر سجل هو الذي يحتوي
     * على آخر حوافز وخصومات محفوظة.
     */

    payrollMap[rowTeacherId] = {

      rowNumber:
        i + 1,


      transactionId:
        payrollTransactionCol !== -1
          ? row[payrollTransactionCol]
          : '',


      bonus:
        payrollBonusCol !== -1
          ? Number(
              row[payrollBonusCol]
            ) || 0
          : 0,


      deduction:
        payrollDeductionCol !== -1
          ? Number(
              row[payrollDeductionCol]
            ) || 0
          : 0,


      status:
        payrollStatusCol !== -1
          ? String(
              row[payrollStatusCol] ||
              'لم يتم'
            )
          : 'لم يتم'

    };

  }


  /* ==========================
     حساب الحصص لكل مدرس
  ========================== */

  const teacherStats = {};


  for (
    let i = 1;
    i < lessonsData.length;
    i++
  ) {

    const row =
      lessonsData[i];


    const teacherId =
      String(
        row[lessonTeacherIdCol] || ''
      ).trim();


    if (!teacherId) {
      continue;
    }


    const lessonDate =
      row[lessonDateCol];


    if (
      !isDateInPayrollMonth(
        lessonDate,
        targetMonth
      )
    ) {

      continue;

    }


    /*
     * لا نحسب إلا الحصة التي اكتمل تقريرها
     */

    let completed =
      false;


    if (
      lessonReportCol !== -1
    ) {

      const reportValue =
        String(
          row[lessonReportCol] || ''
        ).trim();


      completed =
        reportValue !== '';

    }


    /*
     * احتياطًا إذا كانت الحصة مكتملة
     */

    if (
      !completed &&
      lessonStatusCol !== -1
    ) {

      const status =
        String(
          row[lessonStatusCol] || ''
        ).trim();


      completed =
        status === 'مكتملة';

    }


    if (!completed) {
      continue;
    }


    const duration =
      Number(
        row[lessonDurationCol]
      ) || 0;


    if (!teacherStats[teacherId]) {

      teacherStats[teacherId] = {

        lessons: 0,

        minutes: 0

      };

    }


    teacherStats[teacherId].lessons += 1;


    teacherStats[teacherId].minutes +=
      duration;

  }


  /* ==========================
     بناء النتيجة النهائية
  ========================== */

  const result = [];


  for (
    let i = 1;
    i < teachersData.length;
    i++
  ) {

    const teacherRow =
      teachersData[i];


    const teacherId =
      String(
        teacherRow[teacherIdCol] || ''
      ).trim();


    if (!teacherId) {
      continue;
    }


    const teacherName =
      String(
        teacherRow[teacherNameCol] || ''
      ).trim();


    const hourlyRate =
      Number(
        teacherRow[hourlyRateCol]
      ) || 0;


    const stats =
      teacherStats[teacherId] || {

        lessons: 0,

        minutes: 0

      };


    const payroll =
      payrollMap[teacherId] || {

        rowNumber: null,

        transactionId: '',

        bonus: 0,

        deduction: 0,

        status: 'لم يتم'

      };


    const hours =
      stats.minutes / 60;


    /*
     * الراتب الأساسي
     */

    const basicSalary =
      hours * hourlyRate;


    /*
     * الإجمالي النهائي
     */

    const total =
      basicSalary +
      Number(payroll.bonus || 0) -
      Number(payroll.deduction || 0);


    result.push({

      transactionId:
        payroll.transactionId || '',


      month:
        targetMonth,


      teacherId:
        teacherId,


      teacherName:
        teacherName,


      lessons:
        stats.lessons,


      hours:
        Number(
          hours.toFixed(2)
        ),


      minutes:
        stats.minutes,


      hourlyRate:
        hourlyRate,


      bonus:
        Number(
          payroll.bonus || 0
        ),


      deduction:
        Number(
          payroll.deduction || 0
        ),


      total:
        Number(
          total.toFixed(2)
        ),


      status:
        payroll.status ||
        'لم يتم'

    });

  }


  return result;

}




/* ================================
   التحقق من أن التاريخ داخل الشهر
================================ */

function isDateInPayrollMonth(value, month) {

  if (!value) {
    return false;
  }


  let date;


  if (
    Object.prototype.toString.call(value)
    === '[object Date]'
  ) {

    date = value;

  } else {

    date = new Date(value);

  }


  if (isNaN(date.getTime())) {
    return false;
  }


  const dateMonth =
    Utilities.formatDate(
      date,
      Session.getScriptTimeZone(),
      'yyyy-MM'
    );


  return dateMonth === month;

}

/* ================================
   تحديث بيانات راتب المدرس
================================ */

function updatePayrollData_(data) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const sheet =
    ss.getSheetByName('الرواتب');

  if (!sheet) {
    throw new Error(
      'لم يتم العثور على ورقة "الرواتب"'
    );
  }


  const month =
    String(data.month || '').trim();

  const teacherId =
    String(data.teacherId || '').trim();


  if (!month) {
    throw new Error(
      'لم يتم تحديد الشهر'
    );
  }


  if (!teacherId) {
    throw new Error(
      'لم يتم تحديد رقم المدرس'
    );
  }


  /*
   * =====================================================
   * 1) بيانات المدرس الحالية
   * =====================================================
   */

  const payrollData =
    getPayroll_(month);


  const teacherPayroll =
    payrollData.find(function(item) {

      return String(item.teacherId || '').trim()
        === teacherId;

    });


  if (!teacherPayroll) {

    throw new Error(
      'لم يتم العثور على بيانات راتب هذا المدرس لهذا الشهر.'
    );

  }


  const teacherName =
    teacherPayroll.teacherName || '';


  const lessons =
    Number(teacherPayroll.lessons) || 0;


  const hours =
    Number(teacherPayroll.hours) || 0;


  const hourlyRate =
    Number(teacherPayroll.hourlyRate) || 0;


  /*
   * =====================================================
   * 2) أعمدة الشيت
   * =====================================================
   */

  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getValues()[0];


  function getCol(name) {

    return headers.indexOf(name);

  }


  const colTransactionId =
    getCol('رقم العملية');

  const colMonth =
    getCol('الشهر');

  const colTeacherId =
    getCol('رقم المدرس');

  const colTeacherName =
    getCol('اسم المدرس');

  const colLessons =
    getCol('عدد الحصص');

  const colHours =
    getCol('إجمالي الساعات');

  const colHourlyRate =
    getCol('أجر الساعة');

  const colBonus =
    getCol('الحوافز');

  const colDeduction =
    getCol('الخصومات');

  const colTotal =
    getCol('إجمالي المستحق');

  const colStatus =
    getCol('الحالة');


  /*
   * =====================================================
   * 3) البحث عن نفس سجل الراتب
   *
   * الأولوية لرقم العملية
   * لأنه يضمن أننا نعدل نفس الصف
   * =====================================================
   */

  let targetRow = -1;

  let oldBonus = 0;

  let oldDeduction = 0;

  let oldStatus = 'لم يتم';

  let oldTransactionId = '';


  const lastRow =
    sheet.getLastRow();


  if (lastRow > 1) {

    const rows =
      sheet
        .getRange(
          2,
          1,
          lastRow - 1,
          sheet.getLastColumn()
        )
        .getValues();


    /*
     * الرقم الخاص بسجل الراتب الحالي
     */
    const currentTransactionId =
      teacherPayroll.transactionId
        ? String(
            teacherPayroll.transactionId
          ).trim()
        : '';


    /*
     * ---------------------------------------------
     * البحث الأول:
     * باستخدام رقم العملية
     * ---------------------------------------------
     */

    if (
      currentTransactionId &&
      colTransactionId !== -1
    ) {

      for (
        let i = 0;
        i < rows.length;
        i++
      ) {

        const row =
          rows[i];


        const rowTransactionId =
          String(
            row[colTransactionId] || ''
          ).trim();


        if (
          rowTransactionId ===
          currentTransactionId
        ) {

          targetRow = i + 2;

          break;

        }

      }

    }


    /*
     * ---------------------------------------------
     * إذا لم نجد رقم العملية
     * نبحث بالمدرس + الشهر
     * ---------------------------------------------
     */

    if (targetRow === -1) {

      for (
        let i = 0;
        i < rows.length;
        i++
      ) {

        const row =
          rows[i];


        const rowMonth =
          colMonth !== -1
            ? String(
                row[colMonth] || ''
              ).trim()
            : '';


        const rowTeacherId =
          colTeacherId !== -1
            ? String(
                row[colTeacherId] || ''
              ).trim()
            : '';


        if (
          rowMonth === month &&
          rowTeacherId === teacherId
        ) {

          targetRow = i + 2;

          break;

        }

      }

    }


    /*
     * ---------------------------------------------
     * الآن نقرأ القيم القديمة من نفس الصف
     * ---------------------------------------------
     */

    if (targetRow !== -1) {

      const oldRow =
        sheet
          .getRange(
            targetRow,
            1,
            1,
            sheet.getLastColumn()
          )
          .getValues()[0];


      if (colBonus !== -1) {

        oldBonus =
          Number(
            oldRow[colBonus]
          ) || 0;

      }


      if (colDeduction !== -1) {

        oldDeduction =
          Number(
            oldRow[colDeduction]
          ) || 0;

      }


      if (colStatus !== -1) {

        oldStatus =
          oldRow[colStatus] ||
          'لم يتم';

      }


      if (colTransactionId !== -1) {

        oldTransactionId =
          oldRow[colTransactionId] ||
          '';

      }

    }

  }


  /*
   * =====================================================
   * 4) الإضافات الجديدة فقط
   * =====================================================
   */

  const addedBonus =
    Number(data.bonus) || 0;


  const addedDeduction =
    Number(data.deduction) || 0;


  /*
   * =====================================================
   * 5) التراكم
   * =====================================================
   */

  const totalBonus =
    oldBonus + addedBonus;


  const totalDeduction =
    oldDeduction + addedDeduction;


  /*
   * =====================================================
   * 6) إجمالي الراتب
   * =====================================================
   */

  const basicSalary =
    hours * hourlyRate;


  const total =
    basicSalary +
    totalBonus -
    totalDeduction;


  /*
   * =====================================================
   * 7) إذا لم نجد السجل فعليًا
   * ننشئه
   * =====================================================
   */

  if (targetRow === -1) {

    targetRow =
      sheet.getLastRow() + 1;

  }


  const rowData =
    new Array(
      sheet.getLastColumn()
    ).fill('');


  /*
   * رقم العملية
   */

  if (colTransactionId !== -1) {

    rowData[colTransactionId] =
      oldTransactionId ||
      teacherPayroll.transactionId ||
      generatePayrollTransactionId_(
        sheet
      );

  }


  /*
   * الشهر
   */

  if (colMonth !== -1) {

    rowData[colMonth] =
      month;

  }


  /*
   * رقم المدرس
   */

  if (colTeacherId !== -1) {

    rowData[colTeacherId] =
      teacherId;

  }


  /*
   * اسم المدرس
   */

  if (colTeacherName !== -1) {

    rowData[colTeacherName] =
      teacherName;

  }


  /*
   * عدد الحصص
   */

  if (colLessons !== -1) {

    rowData[colLessons] =
      lessons;

  }


  /*
   * إجمالي الساعات
   */

  if (colHours !== -1) {

    rowData[colHours] =
      hours;

  }


  /*
   * أجر الساعة
   */

  if (colHourlyRate !== -1) {

    rowData[colHourlyRate] =
      hourlyRate;

  }


  /*
   * إجمالي الحوافز
   */

  if (colBonus !== -1) {

    rowData[colBonus] =
      totalBonus;

  }


  /*
   * إجمالي الخصومات
   */

  if (colDeduction !== -1) {

    rowData[colDeduction] =
      totalDeduction;

  }


  /*
   * إجمالي المستحق
   */

  if (colTotal !== -1) {

    rowData[colTotal] =
      total;

  }


  /*
   * حالة التحويل
   */

  if (colStatus !== -1) {

    rowData[colStatus] =
      data.status ||
      oldStatus ||
      'لم يتم';

  }


  /*
   * =====================================================
   * 8) حفظ نفس السجل
   * =====================================================
   */

  sheet
    .getRange(
      targetRow,
      1,
      1,
      rowData.length
    )
    .setValues([rowData]);


  /*
   * =====================================================
   * 9) إرجاع البيانات الجديدة للواجهة
   * =====================================================
   */

  return payrollData.map(function(item) {

    if (
      String(item.teacherId || '').trim()
      !== teacherId
    ) {

      return item;

    }


    return {

      transactionId:
        rowData[colTransactionId],

      month:
        month,

      teacherId:
        teacherId,

      teacherName:
        teacherName,

      lessons:
        lessons,

      hours:
        hours,

      minutes:
        Number(item.minutes) || 0,

      hourlyRate:
        hourlyRate,

      bonus:
        totalBonus,

      deduction:
        totalDeduction,

      total:
        total,

      status:
        data.status ||
        oldStatus ||
        'لم يتم'

    };

  });

}




/* ================================
   مزامنة قائمة الرواتب
================================ */

function syncPayrollTeachers_(month) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const teachersSheet =
    ss.getSheetByName('المدرسون');

  const payrollSheet =
    ss.getSheetByName('الرواتب');


  if (!teachersSheet) {
    throw new Error(
      'لم يتم العثور على ورقة المدرسين.'
    );
  }

  if (!payrollSheet) {
    throw new Error(
      'لم يتم العثور على ورقة الرواتب.'
    );
  }


  /*
   * ==========================
   * توحيد صيغة الشهر
   * ==========================
   */

  function normalizeMonth(value) {

    if (!value) {
      return '';
    }


    // إذا كانت قيمة Date فعلية
    if (
      Object.prototype.toString.call(value)
      === '[object Date]'
    ) {

      if (isNaN(value.getTime())) {
        return '';
      }

      return Utilities.formatDate(
        value,
        Session.getScriptTimeZone(),
        'yyyy-MM'
      );

    }


    const text =
      String(value).trim();


    if (!text) {
      return '';
    }


    // صيغة yyyy-MM
    if (
      /^\d{4}-\d{2}$/.test(text)
    ) {
      return text;
    }


    // صيغة yyyy-MM-dd
    if (
      /^\d{4}-\d{2}-\d{2}$/.test(text)
    ) {
      return text.substring(0, 7);
    }


    // محاولة قراءة التاريخ كنص
    const parsed =
      new Date(text);


    if (!isNaN(parsed.getTime())) {

      return Utilities.formatDate(
        parsed,
        Session.getScriptTimeZone(),
        'yyyy-MM'
      );

    }


    return text;
  }


  /*
   * ==========================
   * الشهر المطلوب
   * ==========================
   */

  if (!month) {

    month =
      Utilities.formatDate(
        new Date(),
        Session.getScriptTimeZone(),
        'yyyy-MM'
      );

  }


  month =
    normalizeMonth(month);


  if (!month) {
    throw new Error(
      'تعذر تحديد الشهر.'
    );
  }


  /*
   * ==========================
   * قراءة البيانات
   * ==========================
   */

  const teachersData =
    teachersSheet
      .getDataRange()
      .getValues();


  const payrollData =
    payrollSheet
      .getDataRange()
      .getValues();


  if (teachersData.length < 2) {
    return true;
  }


  if (payrollData.length < 1) {
    throw new Error(
      'ورقة الرواتب لا تحتوي على صف العناوين.'
    );
  }


  /*
   * ==========================
   * العناوين
   * ==========================
   */

  const teacherHeaders =
    teachersData[0].map(function(h) {
      return String(h).trim();
    });


  const payrollHeaders =
    payrollData[0].map(function(h) {
      return String(h).trim();
    });


  /*
   * ==========================
   * أعمدة المدرسين
   * ==========================
   */

  const teacherIdCol =
    teacherHeaders.indexOf('رقم المدرس');

  const teacherNameCol =
    teacherHeaders.indexOf('اسم المدرس');


  if (
    teacherIdCol === -1 ||
    teacherNameCol === -1
  ) {
    throw new Error(
      'تأكد من وجود عمودي رقم المدرس واسم المدرس في ورقة المدرسين.'
    );
  }


  /*
   * ==========================
   * أعمدة الرواتب
   * ==========================
   */

  const payrollMonthCol =
    payrollHeaders.indexOf('الشهر');

  const payrollTeacherIdCol =
    payrollHeaders.indexOf('رقم المدرس');

  const payrollTeacherNameCol =
    payrollHeaders.indexOf('اسم المدرس');

  const payrollLessonsCol =
    payrollHeaders.indexOf('عدد الحصص');

  const payrollHoursCol =
    payrollHeaders.indexOf('إجمالي الساعات');

  const payrollRateCol =
    payrollHeaders.indexOf('أجر الساعة');

  const payrollBonusCol =
    payrollHeaders.indexOf('الحوافز');

  const payrollDeductionCol =
    payrollHeaders.indexOf('الخصومات');

  const payrollTotalCol =
    payrollHeaders.indexOf('إجمالي المستحق');

  const payrollStatusCol =
    payrollHeaders.indexOf(
      'الحالة ( تم الدفع/ لم يتم)'
    );

  const payrollTransactionCol =
    payrollHeaders.indexOf('رقم العملية');


  if (
    payrollMonthCol === -1 ||
    payrollTeacherIdCol === -1
  ) {
    throw new Error(
      'تأكد من وجود عمودي الشهر ورقم المدرس في ورقة الرواتب.'
    );
  }


  /*
   * ==========================
   * المدرسون الموجودون بالفعل
   * ==========================
   *
   * مهم جدًا:
   * نستخدم normalizeMonth()
   * بدل المقارنة النصية المباشرة.
   */

  const existing = {};


  for (
    let i = 1;
    i < payrollData.length;
    i++
  ) {

    const row =
      payrollData[i];


    const rowMonth =
      normalizeMonth(
        row[payrollMonthCol]
      );


    const teacherId =
      String(
        row[payrollTeacherIdCol] || ''
      ).trim();


    if (
      rowMonth === month &&
      teacherId
    ) {

      existing[teacherId] = true;

    }

  }


  /*
   * ==========================
   * إضافة المدرسين الجدد فقط
   * ==========================
   */

  for (
    let i = 1;
    i < teachersData.length;
    i++
  ) {

    const teacherRow =
      teachersData[i];


    const teacherId =
      String(
        teacherRow[teacherIdCol] || ''
      ).trim();


    const teacherName =
      String(
        teacherRow[teacherNameCol] || ''
      ).trim();


    if (!teacherId) {
      continue;
    }


    /*
     * إذا كان للمدرس سجل بالفعل
     * لن ننشئ سجلًا جديدًا.
     *
     * وبالتالي لن نلمس:
     * الحوافز
     * الخصومات
     * حالة الدفع
     */

    if (existing[teacherId]) {
      continue;
    }


    /*
     * ==========================
     * إنشاء سجل جديد
     * ==========================
     */

    const newRow =
      new Array(
        payrollHeaders.length
      ).fill('');


    if (
      payrollTransactionCol !== -1
    ) {

      newRow[payrollTransactionCol] =
        generatePayrollTransactionId_(
          payrollSheet
        );

    }


    if (
      payrollMonthCol !== -1
    ) {

      newRow[payrollMonthCol] =
        month;

    }


    if (
      payrollTeacherIdCol !== -1
    ) {

      newRow[payrollTeacherIdCol] =
        teacherId;

    }


    if (
      payrollTeacherNameCol !== -1
    ) {

      newRow[payrollTeacherNameCol] =
        teacherName;

    }


    if (
      payrollLessonsCol !== -1
    ) {

      newRow[payrollLessonsCol] =
        0;

    }


    if (
      payrollHoursCol !== -1
    ) {

      newRow[payrollHoursCol] =
        0;

    }


    if (
      payrollRateCol !== -1
    ) {

      newRow[payrollRateCol] =
        '';

    }


    if (
      payrollBonusCol !== -1
    ) {

      newRow[payrollBonusCol] =
        0;

    }


    if (
      payrollDeductionCol !== -1
    ) {

      newRow[payrollDeductionCol] =
        0;

    }


    if (
      payrollTotalCol !== -1
    ) {

      newRow[payrollTotalCol] =
        0;

    }


    if (
      payrollStatusCol !== -1
    ) {

      newRow[payrollStatusCol] =
        'لم يتم';

    }


    payrollSheet.appendRow(
      newRow
    );

  }


  return true;

}


/* ================================
   رقم عملية الراتب
================================ */

function generatePayrollTransactionId_(sheet) {

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return 'PAY-0001';
  }


  const headers =
    sheet
      .getRange(1, 1, 1, sheet.getLastColumn())
      .getValues()[0];


  const idCol =
    headers.indexOf('رقم العملية');


  if (idCol === -1) {
    return 'PAY-' + Date.now();
  }


  const values =
    sheet
      .getRange(
        2,
        idCol + 1,
        lastRow - 1,
        1
      )
      .getValues();


  let maxNumber = 0;


  values.forEach(function(row) {

    const value =
      String(row[0] || '');

    const match =
      value.match(/PAY-(\d+)/);


    if (match) {

      const number =
        Number(match[1]);

      if (number > maxNumber) {
        maxNumber = number;
      }

    }

  });


  return 'PAY-' +
    String(maxNumber + 1).padStart(4, '0');

}

function createUniqueLessonId_(
  existingData,
  headers,
  generatedLessonIds
) {

  const lessonIdCol =
    headers.indexOf('رقم الحصة');

  let maxNumber = 0;

  /*
   * البحث عن أكبر رقم حصة موجود حاليًا
   */

  if (lessonIdCol !== -1) {

    for (
      let i = 1;
      i < existingData.length;
      i++
    ) {

      const existingId =
        String(
          existingData[i][lessonIdCol] || ''
        ).trim();

      /*
       * مثال:
       * cla-003
       * cla-025
       * cla-104
       */

      const match =
        existingId.match(/(\d+)$/);

      if (match) {

        const number =
          Number(match[1]);

        if (
          !isNaN(number) &&
          number > maxNumber
        ) {

          maxNumber = number;

        }

      }

    }

  }


  /*
   * البحث أيضًا في الأرقام التي تم
   * إنشاؤها أثناء العملية الحالية
   */

  Object.keys(generatedLessonIds || {}).forEach(
    function(id) {

      const match =
        String(id).match(/(\d+)$/);

      if (match) {

        const number =
          Number(match[1]);

        if (
          !isNaN(number) &&
          number > maxNumber
        ) {

          maxNumber = number;

        }

      }

    }
  );


  /*
   * إنشاء الرقم التالي
   */

  let nextNumber =
    maxNumber + 1;


  let lessonId =
    'cla-' +
    String(nextNumber).padStart(3, '0');


  /*
   * ضمان إضافي لعدم التكرار
   */

  while (
    generatedLessonIds &&
    generatedLessonIds[lessonId]
  ) {

    nextNumber++;

    lessonId =
      'cla-' +
      String(nextNumber).padStart(3, '0');

  }


  return lessonId;

}

function normName_(s) {
  return String(s || '')
    .replace(/[\u064B-\u065F\u0640]/g, '')   // تشكيل وتطويل
    .replace(/[أإآ]/g, 'ا')
    .replace(/ى/g, 'ي')
    .replace(/ة/g, 'ه')
    .replace(/\s+/g, ' ')
    .trim();
}

function plain_(v) {
  if (v instanceof Date) {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return v === null || v === undefined ? '' : v;
}

function getTeacherStudentsDirect_(teacherId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const teachersSheet = ss.getSheetByName('المدرسون');
  const studentsSheet = ss.getSheetByName('الطلاب');
  if (!teachersSheet) throw new Error('ورقة المدرسون غير موجودة');
  if (!studentsSheet) throw new Error('ورقة الطلاب غير موجودة');

  const tLast = teachersSheet.getLastRow();
  if (tLast < 2) return { teacher: null, students: [] };

  const tRows = teachersSheet.getRange(2, 1, tLast - 1, 9).getValues();
  let teacher = null;
  for (let i = 0; i < tRows.length; i++) {
    const r = tRows[i];
    if (String(r[0] || '').trim() === String(teacherId || '').trim()) {
      teacher = {
        id: r[0],
        name: String(r[1] || '').trim(),
        whatsapp: plain_(r[2]),
        specialization: plain_(r[3]),
        hourlyRate: Number(r[4]) || 0,
        joinDate: formatDate(r[5]),
        status: plain_(r[6]),
        notes: plain_(r[7]),
        age: plain_(r[8])
      };
      break;
    }
  }
  if (!teacher) throw new Error('لم يتم العثور على المعلم بالرقم: ' + teacherId);

  const students = [];
  const sLast = studentsSheet.getLastRow();
  if (sLast >= 2) {
    const n = sLast - 1;
    const rows = studentsSheet.getRange(2, 1, n, 17).getValues();
    // وقت الحصة بالشكل المعروض في الشيت بالظبط، عشان نتجنب مشاكل Date
    const timeShown = studentsSheet.getRange(2, 12, n, 1).getDisplayValues();
    const target = normName_(teacher.name);

    for (let i = 0; i < rows.length; i++) {
      const r = rows[i];
      if (!r[0]) continue;
      if (normName_(r[6]) !== target) continue;

      students.push({
        id: String(r[0]),
        name: plain_(r[1]),
        whatsapp: plain_(r[2]),
        birthDate: safeFormatStudentDate(r[3]),
        program: plain_(r[4]),
        level: plain_(r[5]),
        teacher: plain_(r[6]),
        supervisor: plain_(r[7]),
        monthlyLessons: plain_(r[8]),
        lessonDuration: plain_(r[9]),
        lessonDays: plain_(r[10]),
        lessonTime: timeShown[i][0],
        lessonLink: plain_(r[12]),
        subscriptionStart: safeFormatStudentDate(r[13]),
        status: plain_(r[14]),
        notes: plain_(r[15]),
        subscriptionPrice: Number(r[16]) || 0
      });
    }
  }
  return { teacher: teacher, students: students };
}

// تحويل الوقت المعروض في الشيت إلى HH:mm (24 ساعة)
function timeToHHmm_(display) {
  const s = String(display || '').trim();
  const m = s.match(/(\d{1,2}):(\d{2})/);
  if (!m) return '';
  let h = Number(m[1]);
  if (/pm|م/i.test(s) && h < 12) h += 12;
  if (/am|ص/i.test(s) && h === 12) h = 0;
  return ('0' + h).slice(-2) + ':' + m[2];
}

function hashPassword_(password, salt) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    salt + ':' + password,
    Utilities.Charset.UTF_8
  );
  return 'v1:' + Utilities.base64Encode(bytes);
}

function login(username, password) {
  username = String(username || '').trim().toLowerCase();
  password = String(password || '');
  if (!username || !password) {
    throw new Error('أدخل اسم المستخدم وكلمة المرور');
  }

  const cache = CacheService.getScriptCache();
  const failKey = 'fail:' + username;
  const fails = Number(cache.get(failKey)) || 0;
  if (fails >= 5) {
    throw new Error('تم إيقاف المحاولات مؤقتًا، حاول بعد 15 دقيقة');
  }

  const sheet = getUsersSheet_();
  const last = sheet.getLastRow();

  if (last >= 2) {
    const rows = sheet.getRange(2, 1, last - 1, 8).getValues();
    for (let i = 0; i < rows.length; i++) {
      const r = rows[i];
      if (String(r[0]).trim().toLowerCase() !== username) continue;
      if (String(r[6]).trim() !== 'نشط') break;

      if (hashPassword_(password, String(r[2])) === String(r[1])) {
        cache.remove(failKey);
        sheet.getRange(i + 2, 8).setValue(new Date());

        const user = {
          username: String(r[0]),
          role: String(r[3]),
          linkedId: String(r[4]),
          name: String(r[5]),
          iat: Date.now()
        };
        const token = Utilities.getUuid() + Utilities.getUuid();
        cache.put('sess:' + token, JSON.stringify(user), 21600);
        return { token: token, user: user };
      }
      break;
    }
  }

  cache.put(failKey, String(fails + 1), 900);
  throw new Error('اسم المستخدم أو كلمة المرور غير صحيحة');
}

function logout(token) {
  CacheService.getScriptCache().remove('sess:' + token);
  return true;
}

function requireAuth_(token, allowedRoles) {
  if (!token) throw new Error('SESSION_EXPIRED');

  const cache = CacheService.getScriptCache();
  const raw = cache.get('sess:' + token);
  if (!raw) throw new Error('SESSION_EXPIRED');

  const user = JSON.parse(raw);

  // لو الحساب اتعدّل أو اتحذف بعد ما الجلسة اتفتحت، الجلسة تنتهي
  const revokedAt = Number(cache.get('revoked:' + user.username)) || 0;
  if (revokedAt > (user.iat || 0)) {
    cache.remove('sess:' + token);
    throw new Error('SESSION_EXPIRED');
  }

  if (allowedRoles && allowedRoles.indexOf(user.role) === -1) {
    throw new Error('ليس لديك صلاحية لهذا الإجراء');
  }
  return user;
}

function saveUser(token, data) {
  const me = requireAuth_(token, ['admin']);
  data = data || {};

  const mode = data.mode === 'edit' ? 'edit' : 'create';
  const username = String(data.username || '').trim().toLowerCase();
  const role = String(data.role || '').trim();
  const password = String(data.password || '');
  let linkedId = String(data.linkedId || '').trim();
  let name = String(data.name || '').trim();
  const status = mode === 'create' ? 'نشط' : String(data.status || '').trim();

  if (!/^[a-z][a-z0-9._-]{2,29}$/.test(username)) {
    throw new Error(
      'اسم المستخدم: حروف إنجليزية صغيرة وأرقام فقط، يبدأ بحرف، من 3 إلى 30 حرف'
    );
  }
  if (['admin', 'teacher', 'student'].indexOf(role) === -1) {
    throw new Error('الدور غير صحيح');
  }
  if (['نشط', 'موقوف'].indexOf(status) === -1) {
    throw new Error('الحالة غير صحيحة');
  }

  if (role === 'admin') {
    linkedId = '';
  } else {
    if (!linkedId) {
      throw new Error(
        role === 'teacher'
          ? 'اختر المدرس المرتبط بالحساب'
          : 'اختر الطالب المرتبط بالحساب'
      );
    }
    const person = findLinkedPerson_(role, linkedId);
    if (!person) throw new Error('الرقم المرتبط غير موجود في الشيت');
    if (!name) name = person.name;
  }
  if (!name) name = username;

  const sheet = getUsersSheet_();
  const found = findUserRow_(sheet, username);

  /* ---------- إنشاء ---------- */
  if (mode === 'create') {
    if (found) throw new Error('اسم المستخدم موجود بالفعل');
    addUser_(username, password, role, linkedId, name);
    return { success: true, message: 'تم إنشاء الحساب بنجاح' };
  }

  /* ---------- تعديل ---------- */
  if (!found) throw new Error('الحساب غير موجود');

  const isSelf =
    username === String(me.username).trim().toLowerCase();

  if (isSelf && (role !== 'admin' || status !== 'نشط')) {
    throw new Error('لا يمكنك تغيير دورك أو إيقاف حسابك الشخصي');
  }

  if (password !== '' && password.length < 8) {
    throw new Error('كلمة المرور لازم تكون 8 حروف على الأقل');
  }

  // D: الدور | E: الرقم المرتبط | F: الاسم | G: الحالة
  sheet.getRange(found.row, 4, 1, 4).setValues([[role, linkedId, name, status]]);

  if (password !== '') {
    const salt = Utilities.getUuid() + Utilities.getUuid();
    sheet.getRange(found.row, 2, 1, 2)
      .setValues([[hashPassword_(password, salt), salt]]);
  }

  // أي تعديل على حساب شخص تاني ينهي جلساته فورًا
  if (!isSelf || password !== '') {
    revokeUserSessions_(username);
  }

  return { success: true, message: 'تم حفظ التعديلات' };
}

function deleteUser(token, username) {
  const me = requireAuth_(token, ['admin']);
  const target = String(username || '').trim().toLowerCase();

  if (!target) throw new Error('اسم المستخدم مطلوب');

  if (target === String(me.username).trim().toLowerCase()) {
    throw new Error('لا يمكنك حذف حسابك الشخصي');
  }

  const sheet = getUsersSheet_();
  const found = findUserRow_(sheet, target);
  if (!found) throw new Error('الحساب غير موجود');

  sheet.deleteRow(found.row);
  revokeUserSessions_(target);

  return { success: true, message: 'تم حذف الحساب' };
}

// توحيد شكل الشهر إلى yyyy-MM (نفس منطق getPayroll)
function normPayMonth_(value) {
  if (!value) return '';

  if (Object.prototype.toString.call(value) === '[object Date]') {
    if (isNaN(value.getTime())) return '';
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM');
  }

  const text = String(value).trim().replace(/\//g, '-');
  const m = text.match(/^(\d{4})-(\d{1,2})/);
  return m ? m[1] + '-' + ('0' + m[2]).slice(-2) : text;
}

}

function findPayrollStatusCol_(headers) {
  let c = headers.indexOf('الحالة');
  if (c === -1) c = headers.indexOf('الحالة ( تم الدفع/ لم يتم)');
  if (c === -1) {
    c = headers.findIndex(function(h) {
      return String(h).trim().indexOf('الحالة') === 0;
    });
  }
  return c;
}

// آخر سجل للمدرس في الشهر (نفس منطق getPayroll: آخر سجل هو المعتمد)
function findPayrollRow_(sheet, month, teacherId) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 1) {
    throw new Error('ورقة الرواتب لا تحتوي على صف العناوين.');
  }

  const headers = data[0].map(function(h) { return String(h).trim(); });
  const colMonth = headers.indexOf('الشهر');
  const colTeacher = headers.indexOf('رقم المدرس');
  const colStatus = findPayrollStatusCol_(headers);

  if (colMonth === -1 || colTeacher === -1) {
    throw new Error('تأكد من وجود عمودي الشهر ورقم المدرس في ورقة الرواتب.');
  }

  let found = null;
  for (let i = 1; i < data.length; i++) {
    if (normPayMonth_(data[i][colMonth]) !== month) continue;
    if (String(data[i][colTeacher] || '').trim() !== teacherId) continue;

    found = {
      rowNumber: i + 1,
      status: colStatus !== -1 ? String(data[i][colStatus] || '').trim() : ''
    };
  }

  return { headers: headers, colStatus: colStatus, row: found };
}

// نفس معادلة صفحة الرواتب بالظبط (الساعات مقرّبة × أجر الساعة + الحوافز - الخصومات)
function payrollBasic_(entry) {
  return (Number(entry.hours) || 0) * (Number(entry.hourlyRate) || 0);
}
function payrollTotal_(entry) {
  return payrollBasic_(entry) +
    (Number(entry.bonus) || 0) -
    (Number(entry.deduction) || 0);
}

// إضافة علامات الإغلاق لنتيجة الرواتب (من غير المساس بالأرقام)
function annotatePayroll_(rows) {
  const cur = getCurrentMonthKey_();
  (rows || []).forEach(function(r) {
    r.locked = r.month !== cur;   // مقفل: أي شهر غير الشهر الحالي
    r.closed = r.month < cur;     // شهر انتهى
  });
  return rows;
}

function markPayrollPaid_(teacherId, month) {
  teacherId = String(teacherId || '').trim();
  month = normPayMonth_(month);

  if (!teacherId) throw new Error('لم يتم تحديد المدرس');
  if (!/^\d{4}-\d{2}$/.test(month)) throw new Error('الشهر غير صحيح');
  if (month > getCurrentMonthKey_()) {
    throw new Error('لا يمكن دفع راتب شهر لم يبدأ بعد');
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('الرواتب');
  if (!sheet) throw new Error('لم يتم العثور على ورقة الرواتب');

  const entry = getPayroll_(month).find(function(i) {
    return String(i.teacherId).trim() === teacherId;
  });
  if (!entry) throw new Error('لم يتم العثور على المدرس');

  const info = findPayrollRow_(sheet, month, teacherId);
  if (info.colStatus === -1) {
    throw new Error('عمود الحالة غير موجود في ورقة الرواتب (لازم عنوانه يبدأ بكلمة "الحالة")');
  }
  if (info.row && isPaidStatus_(info.row.status)) {
    throw new Error('تم دفع هذا الراتب بالفعل');
  }

  const headers = info.headers;
  let rowNumber;

  if (info.row) {
    rowNumber = info.row.rowNumber;
  } else {
    // مفيش سجل للمدرس في الشهر ده: ننشئه (الحوافز والخصومات = 0)
    const newRow = new Array(headers.length).fill('');
    const set = function(name, value) {
      const c = headers.indexOf(name);
      if (c !== -1) newRow[c] = value;
    };
    set('رقم العملية', generatePayrollTransactionId_(sheet));
    set('الشهر', month);
    set('رقم المدرس', teacherId);
    set('اسم المدرس', entry.teacherName);
    set('الحوافز', 0);
    set('الخصومات', 0);
    sheet.appendRow(newRow);
    rowNumber = sheet.getLastRow();
  }

  // نكتب خلايا محددة فقط. الحوافز والخصومات لا تُلمس هنا أبدًا.
  const put = function(name, value) {
    const c = headers.indexOf(name);
    if (c !== -1) sheet.getRange(rowNumber, c + 1).setValue(value);
  };
  const total = payrollTotal_(entry);

  put('عدد الحصص', entry.lessons);
  put('إجمالي الساعات', entry.hours);
  put('أجر الساعة', entry.hourlyRate);
  put('إجمالي المستحق', total);
  sheet.getRange(rowNumber, info.colStatus + 1).setValue('تم');

  SpreadsheetApp.flush();

  return {
    success: true,
    teacherId: teacherId,
    month: month,
    status: 'تم',
    total: total,
    message: 'تم تسجيل دفع الراتب'
  };
}

function getMyPayroll(token) {
  const user = requireAuth_(token, ['teacher']);
  const teacherId = String(user.linkedId || '').trim();
  if (!teacherId) throw new Error('الحساب غير مرتبط بمدرس');

  const tz = Session.getScriptTimeZone();
  const now = new Date();
  const month = getCurrentMonthKey_();

  const entry = getPayroll_(month).find(function(i) {
    return String(i.teacherId).trim() === teacherId;
  });

  // الوقت المتبقي (ملي ثانية) حتى أول لحظة في الشهر الجديد بتوقيت المشروع
  const y = Number(Utilities.formatDate(now, tz, 'yyyy'));
  const mo = Number(Utilities.formatDate(now, tz, 'M'));
  const ny = mo === 12 ? y + 1 : y;
  const nm = mo === 12 ? 1 : mo + 1;
  const nextStart = Utilities.parseDate(
    ny + '-' + ('0' + nm).slice(-2) + '-01 00:00:00',
    tz,
    'yyyy-MM-dd HH:mm:ss'
  );

  return {
    month: month,
    total: entry ? payrollTotal_(entry) : 0,
    bonus: entry ? Number(entry.bonus) || 0 : 0,
    deduction: entry ? Number(entry.deduction) || 0 : 0,
    msToNextMonth: nextStart.getTime() - now.getTime()
  };
}

function withLessonAdminLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    return fn();
  } finally {
    lock.releaseLock();
  }
}

// يضيف الأعمدة الإضافية في آخر ورقة الحصص (مرة واحدة فقط) ويرجع العناوين
function ensureLessonExtraCols_(sheet) {
  const extra = [
    LESSON_ADMIN_FLAG_HEADER_,
    LESSON_ADMIN_ORIG_TEACHER_,
    LESSON_ADMIN_ORIG_DATE_
  ];

  let lastCol = sheet.getLastColumn();
  let headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]
    .map(function (h) { return String(h).trim(); });

  extra.forEach(function (name) {
    if (headers.indexOf(name) === -1) {
      lastCol++;
      sheet.getRange(1, lastCol).setValue(name);
      headers.push(name);
    }
  });

  return headers;
}

function colIndex_(headers, name, fallbackIndex) {
  const i = headers.indexOf(name);
  return i !== -1 ? i : fallbackIndex;
}

// تحميل حصة واحدة بكل ما نحتاجه للتعديل
function adminLoadLesson_(lessonId) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('الحصص');
  if (!sheet) throw new Error('ورقة الحصص غير موجودة');

  const headers = ensureLessonExtraCols_(sheet);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) throw new Error('لا توجد حصص مسجلة');

  const values = sheet.getRange(1, 1, lastRow, headers.length).getValues();

  const col = {
    id:         colIndex_(headers, 'رقم الحصة', 0),
    date:       colIndex_(headers, 'التاريخ', 1),
    time:       colIndex_(headers, 'الوقت', 2),
    studentId:  colIndex_(headers, 'رقم الطالب', 3),
    teacherId:  colIndex_(headers, 'رقم المدرس', 5),
    teacherName:colIndex_(headers, 'اسم المدرس', 6),
    duration:   colIndex_(headers, 'مدة الحصة', 7),
    status:     colIndex_(headers, 'حالة الحصة', 9),
    pay:        colIndex_(headers, 'أجر المدرس', 13),
    flag:       headers.indexOf(LESSON_ADMIN_FLAG_HEADER_),
    origTeacher:headers.indexOf(LESSON_ADMIN_ORIG_TEACHER_),
    origDate:   headers.indexOf(LESSON_ADMIN_ORIG_DATE_)
  };

  const target = String(lessonId || '').trim();

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][col.id]).trim() === target) {
      return {
        sheet: sheet,
        headers: headers,
        values: values,
        index: i,
        rowNumber: i + 1,
        row: values[i],
        col: col
      };
    }
  }

  throw new Error('لم يتم العثور على الحصة');
}

function setLessonCell_(ctx, colIdx, value) {
  ctx.sheet.getRange(ctx.rowNumber, colIdx + 1).setValue(value);
}

  }
}

/* =========================================================
   3) تغيير موعد حصة واحدة فقط (لا يمس جدول الطالب)
========================================================= */

function adminRescheduleLesson(token, lessonId, newDate, newTime) {
  requireAuth_(token, ['admin']);

  return withLessonAdminLock_(function () {

    const dm = String(newDate || '').match(/^(\d{4})-(\d{2})-(\d{2})$/);
    const tm = String(newTime || '').match(/^(\d{1,2}):(\d{2})$/);

    if (!dm) throw new Error('التاريخ الجديد غير صحيح');
    if (!tm) throw new Error('الوقت الجديد غير صحيح');

    const y = Number(dm[1]), mo = Number(dm[2]), d = Number(dm[3]);
    const hh = Number(tm[1]), mm = Number(tm[2]);

    const newLessonDate = new Date(y, mo - 1, d, hh, mm, 0, 0);
    if (isNaN(newLessonDate.getTime())) throw new Error('الموعد الجديد غير صحيح');

    const ctx = adminLoadLesson_(lessonId);
    assertLessonNotStarted_(ctx);

    const tz = Session.getScriptTimeZone();
    const newDateKey = Utilities.formatDate(newLessonDate, tz, 'yyyy-MM-dd');
    const newTimeKey = Utilities.formatDate(newLessonDate, tz, 'HH:mm');

    // حفظ الموعد الأصلي مرة واحدة فقط (أول تغيير)
    const oldDateRaw = ctx.row[ctx.col.date];
    const oldDate = (oldDateRaw instanceof Date) ? oldDateRaw : parseDate(oldDateRaw);
    const oldDateKey = (oldDate && !isNaN(oldDate.getTime()))
      ? Utilities.formatDate(oldDate, tz, 'yyyy-MM-dd') : '';
    const oldTimeKey = getLessonTimeKey_(ctx.row[ctx.col.time]);

    if (oldDateKey === newDateKey && oldTimeKey === newTimeKey) {
      throw new Error('الموعد الجديد مطابق للموعد الحالي');
    }

    if (ctx.col.origDate !== -1 && !String(ctx.row[ctx.col.origDate] || '').trim()) {
      setLessonCell_(ctx, ctx.col.origDate, oldDateKey + ' ' + oldTimeKey);
    }

    setLessonCell_(ctx, ctx.col.date, newLessonDate);
    setLessonCell_(ctx, ctx.col.time, newTimeKey);

    // تنبيه لو المدرس عنده حصة أخرى في نفس الموعد
    const teacherId = String(ctx.row[ctx.col.teacherId] || '').trim();
    let warning = '';

    for (let i = 1; i < ctx.values.length; i++) {
      if (i === ctx.index) continue;

      const r = ctx.values[i];
      if (String(r[ctx.col.teacherId] || '').trim() !== teacherId) continue;
      if (String(r[ctx.col.status] || '').trim() === LESSON_ADMIN_STATUS_EXCUSED_) continue;

      const rd = (r[ctx.col.date] instanceof Date) ? r[ctx.col.date] : parseDate(r[ctx.col.date]);
      if (!rd || isNaN(rd.getTime())) continue;

      if (Utilities.formatDate(rd, tz, 'yyyy-MM-dd') === newDateKey &&
          getLessonTimeKey_(r[ctx.col.time]) === newTimeKey) {
        warning = '\nتنبيه: المدرس لديه حصة أخرى في نفس الموعد (' + String(r[ctx.col.id]) + ')';
        break;
      }
    }

    return {
      success: true,
      message: 'تم تغيير موعد هذه الحصة فقط إلى ' + newDateKey + ' ' + newTimeKey + warning
    };
  });
}


/* =========================================================
   4) تحويل حصة واحدة لمدرس احتياطي (لا يمس جدول الطالب)
   - تظهر في حصص المدرس البديل وتختفي من حصص الأساسي
   - تُحسب في راتب البديل
========================================================= */

function adminReassignLessonTeacher(token, lessonId, newTeacherId) {
  requireAuth_(token, ['admin']);

  return withLessonAdminLock_(function () {

    const newId = String(newTeacherId || '').trim();
    if (!newId) throw new Error('اختر المدرس البديل');

    const newName = getTeacherNameById_(newId);
    if (!newName) throw new Error('لم يتم العثور على المدرس المختار');

    const ctx = adminLoadLesson_(lessonId);
    assertLessonNotStarted_(ctx);

    const currentId = String(ctx.row[ctx.col.teacherId] || '').trim();
    const currentName = String(ctx.row[ctx.col.teacherName] || '').trim();

    if (currentId === newId) {
      throw new Error('هذا المدرس هو نفسه مدرس الحصة الحالي');
    }

    if (ctx.col.origTeacher !== -1) {
      const savedOriginal = String(ctx.row[ctx.col.origTeacher] || '').trim();

      if (!savedOriginal) {
        // أول تحويل: نحفظ المدرس الأصلي
        setLessonCell_(ctx, ctx.col.origTeacher, currentName);
      } else if (savedOriginal === newName) {
        // الرجوع للمدرس الأصلي: نمسح علامة البديل
        setLessonCell_(ctx, ctx.col.origTeacher, '');
      }
    }

    setLessonCell_(ctx, ctx.col.teacherId, newId);
    setLessonCell_(ctx, ctx.col.teacherName, newName);

    return {
      success: true,
      message: 'تم تحويل هذه الحصة فقط إلى المدرس: ' + newName
    };
  });
}


/* =========================================================
   قراءة العلامات الإضافية للحصص (لعرضها في الواجهة)
   متاحة للأدمن والمدرس والطالب، لكن كل واحد يشوف حصصه فقط
========================================================= */

function getLessonFlags(token) {
  const user = requireAuth_(token, ['admin', 'teacher', 'student']);

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('الحصص');
  if (!sheet) return {};

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow < 2 || lastCol < 1) return {};

  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]
    .map(function (h) { return String(h).trim(); });

  const cFlag = headers.indexOf(LESSON_ADMIN_FLAG_HEADER_);
  const cTeacher = headers.indexOf(LESSON_ADMIN_ORIG_TEACHER_);
  const cDate = headers.indexOf(LESSON_ADMIN_ORIG_DATE_);

  if (cFlag === -1 && cTeacher === -1 && cDate === -1) return {};

  // الحصص المسموح للمستخدم يشوفها (نفس منطق getLessons)
  const visible = {};
  getLessons(token).forEach(function (l) { visible[String(l.id).trim()] = true; });

  const cId = colIndex_(headers, 'رقم الحصة', 0);
  const values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
  const result = {};

  values.forEach(function (r) {
    const id = String(r[cId] || '').trim();
    if (!id || !visible[id]) return;

    const flag = cFlag !== -1 ? String(r[cFlag] || '').trim() : '';
    const origTeacher = cTeacher !== -1 ? String(r[cTeacher] || '').trim() : '';
    let origDate = '';
    if (cDate !== -1) {
      // قد يحوّل Google Sheets النص إلى تاريخ تلقائيًا، فنعيد تنسيقه
      origDate = (r[cDate] instanceof Date)
        ? Utilities.formatDate(r[cDate], Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm')
        : String(r[cDate] || '').trim();
    }

    if (!flag && !origTeacher && !origDate) return;

    result[id] = { attendance: flag };

    // المدرس الأصلي والموعد الأصلي لا يظهران للطالب
    if (user.role !== 'student') {
      result[id].originalTeacher = origTeacher;
      result[id].originalDate = origDate;
    }
  });

  return result;
}

   سجل الحصة (للأدمن فقط):
   دخول المعلم / إنهاء الحصة / رفع التقرير

function lessonTlFormat_(value) {
  // يرجع { date:'yyyy-MM-dd', time:'HH:mm:ss', ms:رقم } أو null لو فاضي
  if (value === '' || value === null || value === undefined) return null;

  const tz = Session.getScriptTimeZone();
  let d = value;

  if (!(d instanceof Date)) d = parseDate(value);

  if (!d || isNaN(d.getTime())) {
    return { date: String(value), time: '', ms: null };
  }

  return {
    date: Utilities.formatDate(d, tz, 'yyyy-MM-dd'),
    time: Utilities.formatDate(d, tz, 'HH:mm:ss'),
    ms: d.getTime()
  };
}

function lessonTlMinutes_(fromMs, toMs) {
  if (fromMs === null || toMs === null || fromMs === undefined || toMs === undefined) return null;
  return Math.round((toMs - fromMs) / 60000);
}

function adminGetLessonTimeline(token, lessonId) {
  requireAuth_(token, ['admin']);

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('الحصص');
  if (!sheet) throw new Error('ورقة الحصص غير موجودة');

  const values = sheet.getDataRange().getValues();
  if (values.length < 2) throw new Error('لا توجد حصص مسجلة');

  const headers = values[0].map(function (h) { return String(h).trim(); });

  function ci(name, fallback) {
    const i = headers.indexOf(name);
    return i !== -1 ? i : fallback;
  }

  const cId = ci('رقم الحصة', 0);
  const cDate = ci('التاريخ', 1);
  const cTime = ci('الوقت', 2);
  const cStudent = ci('اسم الطالب', 4);
  const cTeacher = ci('اسم المدرس', 6);
  const cDuration = ci('مدة الحصة', 7);
  const cStatus = ci('حالة الحصة', 9);
  const cStart = ci('وقت الدخول', 10);
  const cEnd = ci('وقت الانتهاء', 11);
  const cReport = ci('تقرير الحصة', 12);
  const cFlag = headers.indexOf('حالة الحضور');

  const target = String(lessonId || '').trim();
  let row = null;

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][cId]).trim() === target) {
      row = values[i];
      break;
    }
  }

  if (!row) throw new Error('لم يتم العثور على الحصة');

  const tz = Session.getScriptTimeZone();

  /* الموعد المقرر (تاريخ + وقت) */
  const scheduledDate = lessonTlFormat_(row[cDate]);
  const timeKey = getLessonTimeKey_(row[cTime]);
  let scheduledMs = null;

  if (scheduledDate && /^\d{4}-\d{2}-\d{2}$/.test(scheduledDate.date)) {
    const dp = scheduledDate.date.split('-').map(Number);
    const tp = String(timeKey).split(':').map(Number);
    const sd = new Date(dp[0], dp[1] - 1, dp[2], tp[0] || 0, tp[1] || 0, 0, 0);
    scheduledMs = isNaN(sd.getTime()) ? null : sd.getTime();
  }

  const start = lessonTlFormat_(row[cStart]);
  const end = lessonTlFormat_(row[cEnd]);

  /* وقت رفع التقرير: من ورقة التقارير حسب رقم التقرير */
  const reportId = String(row[cReport] || '').trim();
  let report = null;

  if (reportId) {
    report = { id: reportId, date: '', time: '', ms: null };

    const reportsSheet = ss.getSheetByName('التقارير');

    if (reportsSheet && reportsSheet.getLastRow() >= 2) {
      const rv = reportsSheet.getDataRange().getValues();
      const rh = rv[0].map(function (h) { return String(h).trim(); });

      let rId = rh.indexOf('رقم التقرير');
      let rCreated = rh.indexOf('وقت إنشاء التقرير');
      if (rId === -1) rId = 0;
      if (rCreated === -1) rCreated = 10;

      for (let i = 1; i < rv.length; i++) {
        if (String(rv[i][rId]).trim() === reportId) {
          const f = lessonTlFormat_(rv[i][rCreated]);
          if (f) {
            report.date = f.date;
            report.time = f.time;
            report.ms = f.ms;
          }
          break;
        }
      }
    }
  }

  const status = String(row[cStatus] || '').trim();
  const attendance = cFlag !== -1 ? String(row[cFlag] || '').trim() : '';

  return {
    lessonId: target,
    studentName: String(row[cStudent] || ''),
    teacherName: String(row[cTeacher] || ''),
    duration: Number(row[cDuration]) || 30,
    status: status,
    attendance: attendance,

    scheduledDate: scheduledDate ? scheduledDate.date : '',
    scheduledTime: timeKey,

    start: start,
    end: end,
    report: report,

    /* موجب = تأخر، سالب = دخل مبكرًا */
    startDiffMin: lessonTlMinutes_(scheduledMs, start ? start.ms : null),
    actualMin: lessonTlMinutes_(start ? start.ms : null, end ? end.ms : null),
    reportDelayMin: lessonTlMinutes_(end ? end.ms : null, report ? report.ms : null)
  };
}
