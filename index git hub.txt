const SECURED = new Set([

function wrapRunner(runner) {
  return new Proxy(runner, {
    get(target, prop) {
      if (prop === 'withSuccessHandler' || prop === 'withUserObject') {
        return function(arg) { return wrapRunner(target[prop](arg)); };
      }
      if (prop === 'withFailureHandler') {
        return function(handler) {
          return wrapRunner(target.withFailureHandler(function(error, obj) {
            if (error && String(error.message || error).indexOf('SESSION_EXPIRED') !== -1) {
              showLogin('انتهت الجلسة، سجّل الدخول من جديد');
              return;
            }
            handler(error, obj);
          }));
        };
      }
      return function() {
        const args = Array.prototype.slice.call(arguments);
        const finalArgs = SECURED.has(prop) ? [SESSION_TOKEN].concat(args) : args;
        return target[prop].apply(target, finalArgs);
      };
    }
  });
}

const server = wrapRunner(google.script.run);

function saveSession_(token, user) {
  SESSION_TOKEN = token;
  CURRENT_USER = user;
  try { sessionStorage.setItem('itqanToken', token); } catch (e) {}
}

function clearSession_() {
  SESSION_TOKEN = null;
  CURRENT_USER = null;
  try { sessionStorage.removeItem('itqanToken'); } catch (e) {}
}

function showLogin(message) {
  stopMyPay_();  
  clearSession_();
  document.getElementById('loginOverlay').style.display = 'flex';
  document.getElementById('userBar').style.display = 'none';
  document.getElementById('loginPassword').value = '';
  document.getElementById('loginError').textContent = message || '';
  document.getElementById('loginBtn').disabled = false;
  document.getElementById('loginBtn').textContent = 'دخول';
  document.getElementById('loginUsername').focus();
}

function startApp_(user,goHome) {
  document.getElementById('loginOverlay').style.display = 'none';
  document.getElementById('loginPassword').value = '';
  document.getElementById('userBarName').textContent =
    user.name || user.username;
  document.getElementById('userBar').style.display = 'flex';
  applyRoleUI_(user.role, goHome);
  if (user.role === 'teacher') {
    startMyPay_();
  } else {
    stopMyPay_();
  }  
  restoreCurrentPage();
}

function doLogin() {
  const btn = document.getElementById('loginBtn');
  if (btn.disabled) return;

  const username = document.getElementById('loginUsername').value.trim();
  const password = document.getElementById('loginPassword').value;
  const errorBox = document.getElementById('loginError');

  if (!username || !password) {
    errorBox.textContent = 'أدخل اسم المستخدم وكلمة المرور';
    return;
  }

  errorBox.textContent = '';
  btn.disabled = true;
  btn.textContent = 'جاري الدخول...';

  google.script.run
    .withSuccessHandler(function(result) {
      btn.disabled = false;
      btn.textContent = 'دخول';
      saveSession_(result.token, result.user);
      startApp_(result.user, true);
    })
    .withFailureHandler(function(error) {
      btn.disabled = false;
      btn.textContent = 'دخول';
      errorBox.textContent =
        String(error.message || error).replace(/^Error:\s*/, '');
    })
    .login(username, password);
}

function doLogout() {
  const token = SESSION_TOKEN;
  document.getElementById('mainContent').innerHTML = '';
  showLogin();
  if (token) {
    google.script.run.logout(token);
  }
}

function initAuth() {
  let token = null;
  try { token = sessionStorage.getItem('itqanToken'); } catch (e) {}

  if (!token) {
    showLogin();
    return;
  }

  google.script.run
    .withSuccessHandler(function(user) {
      SESSION_TOKEN = token;
      CURRENT_USER = user;
      startApp_(user, false);
    })
    .withFailureHandler(function() {
      showLogin('انتهت الجلسة، سجّل الدخول من جديد');
    })
    .checkSession(token);
}

let myPayInterval = null;
let myPayRolloverTimer = null;

function monthLabel_(month) {
  try {
    const parts = String(month).split('-');
    const d = new Date(Number(parts[0]), Number(parts[1]) - 1, 1);
    return d.toLocaleDateString('ar-EG-u-nu-latn', {
      month: 'long',
      year: 'numeric'
    });
  } catch (e) {
    return String(month || '');
  }
}

function setMyPayValue_(id, text) {
  const el = document.getElementById(id);
  if (!el) return;
  if (el.textContent !== text) {
    el.textContent = text;
    el.classList.remove('pay-flash');
    void el.offsetWidth;
    el.classList.add('pay-flash');
  }
}

function scheduleMyPayRollover_(ms) {
  if (myPayRolloverTimer) {
    clearTimeout(myPayRolloverTimer);
    myPayRolloverTimer = null;
  }
  // أول ثانية في الشهر الجديد: نحدّث العداد فورًا
  if (ms > 0 && ms < 2147000000) {
    myPayRolloverTimer = setTimeout(refreshMyPay_, ms + 700);
  }
}

function refreshMyPay_() {
  if (!CURRENT_USER || CURRENT_USER.role !== 'teacher') return;

  server
    .withSuccessHandler(function(p) {
      if (!p) return;
      setMyPayValue_('payAmount', formatMoney(p.total));
      setMyPayValue_('payBonus', formatMoney(p.bonus));
      setMyPayValue_('payDeduction', formatMoney(p.deduction));
      setMyPayValue_('payMonth', 'راتب شهر ' + monthLabel_(p.month));
      scheduleMyPayRollover_(Number(p.msToNextMonth) || 0);
    })
    .withFailureHandler(function() {
      /* نسيب القيمة الحالية ونحاول في الدورة الجاية */
    })
    .getMyPayroll();
}

function startMyPay_() {
  stopMyPay_();
  document.getElementById('payBar').style.display = 'block';
  document.body.classList.add('has-pay-bar');
  refreshMyPay_();
  myPayInterval = setInterval(refreshMyPay_, 60000);
}

function stopMyPay_() {
  if (myPayInterval) { clearInterval(myPayInterval); myPayInterval = null; }
  if (myPayRolloverTimer) { clearTimeout(myPayRolloverTimer); myPayRolloverTimer = null; }
  const bar = document.getElementById('payBar');
  if (bar) bar.style.display = 'none';
  document.body.classList.remove('has-pay-bar');

let payrollLastData = [];

function payrollFinalTotal_(item) {
  // نفس المعادلة المستخدمة دائمًا في الصفحة
  return (Number(item.hours) || 0) * (Number(item.hourlyRate) || 0) +
    (Number(item.bonus) || 0) -
    (Number(item.deduction) || 0);
}

function buildPayrollHtml_(data) {
  if (!data || !data.length) {
    return `
      <div class="empty">
        <div style="font-size:45px;">💰</div>
        <h3>لا يوجد مدرسون</h3>
        <p>سيتم إضافة المدرسين تلقائيًا إلى الرواتب.</p>
      </div>
    `;
  }

  const lockedMonth = data[0].locked === true;
  const closedMonth = data[0].closed === true;

  let html = '';

  if (closedMonth) {
    html += `
      <div class="pr-note">
        🔒 هذا الشهر مقفل: لا يمكن إضافة حوافز أو خصومات،
        ويمكنك فقط دفع الرواتب التي لم تُدفع بعد.
      </div>
    `;
  } else if (lockedMonth) {
    html += `
      <div class="pr-note">
        ⏳ هذا الشهر لم يبدأ بعد.
      </div>
    `;
  }

  html += `
    <div class="table-container">
      <table>
        <thead>
          <tr>
            <th>اسم المدرس</th>
            <th>عدد الحصص</th>
            <th>إجمالي الساعات</th>
            <th>أجر الساعة</th>
            <th>الحوافز</th>
            <th>الخصومات</th>
            <th>إجمالي المستحق</th>
            <th>حالة الدفع</th>
            <th>إجراء</th>
          </tr>
        </thead>
        <tbody>
  `;

  data.forEach(function(item) {
    const id = escapeHtml(item.teacherId);
    const paid = String(item.status || '').trim().indexOf('تم') === 0;
    const locked = item.locked === true;
    const editable = !locked && !paid;

    const totalBonus = Number(item.bonus) || 0;
    const totalDeduction = Number(item.deduction) || 0;
    const basicSalary =
      (Number(item.hours) || 0) * (Number(item.hourlyRate) || 0);
    const finalTotal = basicSalary + totalBonus - totalDeduction;

    const hasActivity =
      (Number(item.lessons) || 0) > 0 ||
      totalBonus !== 0 ||
      totalDeduction !== 0 ||
      finalTotal !== 0;

    /* ---- الحوافز ---- */
    const bonusCell = `
      <div style="display:flex; flex-direction:column; gap:6px; min-width:150px;">
        ${editable ? `
        <input type="number" min="0" placeholder="حافز جديد" value=""
               id="bonus-${id}"
               style="width:110px; padding:7px; border:1px solid #ddd; border-radius:6px;">
        ` : ''}
        <div style="font-size:12px; color:#18864b; font-weight:600; white-space:nowrap;">
          إجمالي الحوافز خلال الشهر: ${formatMoney(totalBonus)}
        </div>
      </div>
    `;

    /* ---- الخصومات ---- */
    const deductionCell = `
      <div style="display:flex; flex-direction:column; gap:6px; min-width:150px;">
        ${editable ? `
        <input type="number" min="0" placeholder="خصم جديد" value=""
               id="deduction-${id}"
               style="width:110px; padding:7px; border:1px solid #ddd; border-radius:6px;">
        ` : ''}
        <div style="font-size:12px; color:#b35b00; font-weight:600; white-space:nowrap;">
          إجمالي الخصومات خلال الشهر: ${formatMoney(totalDeduction)}
        </div>
      </div>
    `;

    /* ---- حالة الدفع ---- */
    const statusCell = paid
      ? '<span class="sp-chip sp-chip-green">تم دفع الراتب</span>'
      : '<span class="sp-chip pr-unpaid">لم يتم دفع الراتب</span>';

    /* ---- الإجراء ---- */
    let actionCell = '—';
    if (paid) {
      actionCell = '<span class="pr-done">✔ تم الدفع</span>';
    } else {
      const buttons = [];
      if (editable) {
        buttons.push(`
          <button type="button" class="btn btn-primary"
                  data-pr-btn="${id}"
                  onclick="savePayrollRow('${id}', '${escapeHtml(item.month)}')">
            💾 حفظ
          </button>
        `);
      }
      if (hasActivity && !(locked && !item.closed)) {
        buttons.push(`
          <button type="button" class="btn btn-primary pr-pay-btn"
                  data-pr-btn="${id}"
                  onclick="payPayrollRow('${id}', '${escapeHtml(item.month)}')">
            💰 دفع
          </button>
        `);
      }
      if (buttons.length) {
        actionCell = '<div style="display:flex; gap:8px; flex-wrap:wrap;">' +
          buttons.join('') + '</div>';
      }
    }

    html += `
      <tr>
        <td><strong>${escapeHtml(item.teacherName)}</strong></td>
        <td>${escapeHtml(item.lessons)}</td>
        <td>${escapeHtml(item.hours)}</td>
        <td>${formatMoney(item.hourlyRate)}</td>
        <td>${bonusCell}</td>
        <td>${deductionCell}</td>
        <td>
          <div style="display:flex; flex-direction:column; gap:4px;">
            <strong style="font-size:16px;">${formatMoney(finalTotal)}</strong>

            </span>
          </div>
        </td>
        <td>${statusCell}</td>
        <td>${actionCell}</td>
      </tr>
    `;
  });

  html += `
        </tbody>
      </table>
    </div>
  `;

  return html;
}

function renderPayroll(data) {
  const container = document.getElementById('payrollTable');
  if (!container) return;

  payrollLastData = data || [];
  container.innerHTML = buildPayrollHtml_(payrollLastData);
}

function setPayrollRowBusy_(teacherId, busy) {
  document
    .querySelectorAll('button[data-pr-btn="' + teacherId + '"]')
    .forEach(function(b) { b.disabled = busy; });
}

function savePayrollRow(teacherId, month) {
  const bonusInput = document.getElementById('bonus-' + teacherId);
  const deductionInput = document.getElementById('deduction-' + teacherId);

  if (!bonusInput || !deductionInput) {
    alert('تعذر قراءة بيانات الراتب.');
    return;
  }

  // الحافز والخصم الجديد فقط (بيتضافوا على اللي محفوظ)
  const addedBonus = Number(bonusInput.value) || 0;
  const addedDeduction = Number(deductionInput.value) || 0;

  if (addedBonus === 0 && addedDeduction === 0) {
    alert('اكتب حافز جديد أو خصم جديد أولًا.');
    return;
  }

  setPayrollRowBusy_(teacherId, true);

  server
    .withSuccessHandler(function(updatedData) {
      renderPayroll(updatedData || []);
      alert('تم تحديث الراتب بنجاح ✅');
    })
    .withFailureHandler(function(error) {
      setPayrollRowBusy_(teacherId, false);
      alert('حدث خطأ: ' + (error.message || String(error)));
    })
    .updatePayrollData({
      teacherId: teacherId,
      month: month,
      bonus: addedBonus,
      deduction: addedDeduction
    });
}

function payPayrollRow(teacherId, month) {
  const bonusInput = document.getElementById('bonus-' + teacherId);
  const deductionInput = document.getElementById('deduction-' + teacherId);

  // حماية: ما ندفعش وفيه حافز/خصم مكتوب ولسه ما اتحفظش
  if (
    (bonusInput && String(bonusInput.value).trim() !== '') ||
    (deductionInput && String(deductionInput.value).trim() !== '')
  ) {
    alert(
      'فيه حافز أو خصم مكتوب ولسه ما اتحفظش.\n' +
      'اضغط "حفظ" الأول (أو امسح الخانة)، وبعدين اضغط "دفع".'
    );
    return;
  }

  const item = (payrollLastData || []).find(function(i) {
    return String(i.teacherId) === String(teacherId);
  });
  const name = item ? item.teacherName : teacherId;
  const total = item ? payrollFinalTotal_(item) : 0;

  if (!confirm(
    'تأكيد دفع راتب المدرس "' + name + '" عن شهر ' + month + '؟\n\n' +
    'الإجمالي: ' + formatMoney(total) + '\n\n' +
    'بعد الدفع لن يمكن تعديل الحوافز أو الخصومات لهذا الشهر.'
  )) {
    return;
  }

  setPayrollRowBusy_(teacherId, true);

  server
    .withSuccessHandler(function() {
      alert('تم تسجيل دفع الراتب ✅');
      loadPayroll();
    })
    .withFailureHandler(function(error) {
      setPayrollRowBusy_(teacherId, false);
      alert('حدث خطأ: ' + (error.message || String(error)));
    })
    .markPayrollPaid(teacherId, month);
}

function spNum_(value) {
  const n = Number(value);
  return isNaN(n) ? '0' : n.toLocaleString('en-US');
}

function spDate_(value) {
  if (!value) return '—';
  try {
    const d = new Date(String(value) + 'T00:00:00');
    if (isNaN(d.getTime())) return String(value);
    return d.toLocaleDateString('ar-EG-u-nu-latn', {
      weekday: 'long',
      year: 'numeric',
      month: 'long',
      day: 'numeric'
    });
  } catch (e) {
    return String(value);
  }
}

function spField_(label, value) {
  const empty =
    value === null || value === undefined || String(value).trim() === '';
  return (
    '<div>' +
      '<div class="sp-field-label">' + escapeHtml(label) + '</div>' +
      '<div class="sp-field-value">' + escapeHtml(empty ? '—' : value) + '</div>' +
    '</div>'
  );
}

function spPayChip_(isPaid) {
  return isPaid
    ? '<span class="sp-chip sp-chip-green">مدفوع</span>'
    : '<span class="sp-chip sp-chip-red">لم يُدفع</span>';
}

function spStateChip_(remaining) {
  return '<span class="sp-chip sp-chip-gray">' +
    (remaining > 0 ? 'جارٍ' : 'منتهي') + '</span>';
}

      </table>
    </div>
  `;

  return html;
}

function buildUsersTableHtml_(list, search, roleFilter, myUsername) {
  const q = String(search || '').trim().toLowerCase();

  const rows = list.filter(function(u) {
    if (roleFilter && u.role !== roleFilter) return false;
    if (!q) return true;
    return (
      String(u.username).toLowerCase().indexOf(q) !== -1 ||
      String(u.name).toLowerCase().indexOf(q) !== -1
    );
  });

  if (!rows.length) {
    return `
      <div class="empty">
        <div style="font-size:40px">👥</div>
        <h3>لا يوجد مستخدمون مطابقون</h3>
      </div>
    `;
  }

  let html = `
    <div class="sp-table-wrap">
      <table>
        <thead>
          <tr>
            <th>اسم المستخدم</th>
            <th>الاسم</th>
            <th>الدور</th>
            <th>المرتبط به</th>
            <th>الحالة</th>
            <th>آخر دخول</th>
            <th>الإجراءات</th>
          </tr>
        </thead>
        <tbody>
  `;

  rows.forEach(function(u) {
    const isMe =
      String(u.username).toLowerCase() === String(myUsername || '').toLowerCase();
    const uname = escapeHtml(u.username);

    html += `
      <tr>
        <td dir="ltr" style="text-align:right">
          <strong>${uname}</strong>
          ${isMe ? '<span class="sp-chip sp-chip-gray">أنت</span>' : ''}
        </td>
        <td>${escapeHtml(u.name || '—')}</td>
        <td>
          <span class="sp-chip us-role-${escapeHtml(u.role)}">
            ${escapeHtml(USER_ROLE_LABELS[u.role] || u.role || '—')}
          </span>
        </td>
        <td dir="ltr" style="text-align:right">${escapeHtml(u.linkedId || '—')}</td>
        <td>
          ${u.status === 'نشط'
            ? '<span class="sp-chip sp-chip-green">نشط</span>'
            : '<span class="sp-chip sp-chip-red">' + escapeHtml(u.status || 'موقوف') + '</span>'}
        </td>
        <td>${escapeHtml(u.lastLogin || '—')}</td>
        <td>
          <button class="btn btn-secondary"
                  onclick="openUserModal('${uname}')">
            ✏️ تعديل
          </button>
          ${isMe ? '' : `
          <button class="btn btn-danger"
                  onclick="deleteUserUI('${uname}')">
            🗑️ حذف
          </button>`}
        </td>
      </tr>
    `;
  });

  html += `
        </tbody>
      </table>
    </div>
  `;
  return html;
}

function refreshUserLinkSelect_(selectedValue) {
  const role = document.getElementById('userRole').value;
  const group = document.getElementById('userLinkGroup');
  const select = document.getElementById('userLinked');
  const label = document.getElementById('userLinkLabel');

  if (role === 'admin') {
    group.style.display = 'none';
    select.innerHTML = '';
    return;
  }

  group.style.display = '';
  label.textContent =
    role === 'teacher' ? 'المدرس المرتبط *' : 'الطالب المرتبط *';

  const list =
    role === 'teacher' ? userLinkOptions.teachers : userLinkOptions.students;

  let html = '<option value="">' +
    (userLinkLoaded ? '— اختر —' : 'جاري التحميل...') + '</option>';

  let found = false;
  list.forEach(function(o) {
    if (o.id === selectedValue) found = true;
    html +=
      '<option value="' + escapeHtml(o.id) + '">' +
      escapeHtml(o.name) + ' (' + escapeHtml(o.id) + ')</option>';
  });

  // لو الرقم المرتبط القديم مبقاش موجود في الشيت
  if (selectedValue && !found) {
    html +=
      '<option value="' + escapeHtml(selectedValue) + '">' +
      escapeHtml(selectedValue) + ' (غير موجود)</option>';
  }

  select.innerHTML = html;
  select.value = selectedValue || '';
}

function onUserRoleChange() {
  refreshUserLinkSelect_('');
}

function onUserLinkChange() {
  const role = document.getElementById('userRole').value;
  const id = document.getElementById('userLinked').value;
  const nameInput = document.getElementById('userDisplayName');

  if (!id || nameInput.value.trim() !== '') return;

  const list =
    role === 'teacher' ? userLinkOptions.teachers : userLinkOptions.students;
  const person = list.find(function(o) { return o.id === id; });
  if (person) nameInput.value = person.name;
}

function openUserModal(username) {
  ensureUserModal_();

  const modal = document.getElementById('userModal');
  const user = username
    ? usersData.find(function(u) { return u.username === username; })
    : null;

  editingUsername = user ? user.username : null;

  document.getElementById('userError').textContent = '';
  document.getElementById('userSaveBtn').disabled = false;
  document.getElementById('userSaveBtn').textContent = '💾 حفظ';

  const usernameInput = document.getElementById('userUsername');
  const roleSelect = document.getElementById('userRole');
  const isMe = user && CURRENT_USER &&
    String(user.username).toLowerCase() ===
    String(CURRENT_USER.username).toLowerCase();

  if (user) {
    document.getElementById('userModalTitle').textContent =
      'تعديل المستخدم';
    usernameInput.value = user.username;
    usernameInput.disabled = true;
    roleSelect.value = user.role;
    roleSelect.disabled = !!isMe;
    document.getElementById('userDisplayName').value = user.name || '';
    document.getElementById('userStatus').value =
      user.status === 'نشط' ? 'نشط' : 'موقوف';
    document.getElementById('userStatus').disabled = !!isMe;
    document.getElementById('userStatusGroup').style.display = '';
    document.getElementById('userPassLabel').textContent =
      'كلمة مرور جديدة (اختياري)';
    document.getElementById('userPassHint').textContent =
      'اتركها فارغة لو مش عاوز تغيّر كلمة المرور.';
    document.getElementById('userPassword').value = '';
    refreshUserLinkSelect_(user.linkedId);
  } else {
    document.getElementById('userModalTitle').textContent =
      'إضافة مستخدم';
    usernameInput.value = '';
    usernameInput.disabled = false;
    roleSelect.value = 'student';
    roleSelect.disabled = false;

    document.getElementById('userStatus').value = 'نشط';
    document.getElementById('userStatus').disabled = false;
    document.getElementById('userStatusGroup').style.display = 'none';
    document.getElementById('userPassLabel').textContent = 'كلمة المرور *';

function closeUserModal() {
  const modal = document.getElementById('userModal');
  if (modal) modal.style.display = 'none';
  editingUsername = null;
}

function generateUserPassword() {
  const chars = 'abcdefghjkmnpqrstuvwxyzABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let out = '';
  try {
    const arr = new Uint32Array(10);
    crypto.getRandomValues(arr);
    arr.forEach(function(n) { out += chars[n % chars.length]; });
  } catch (e) {
    for (let i = 0; i < 10; i++) {
      out += chars[Math.floor(Math.random() * chars.length)];
    }
  }
  document.getElementById('userPassword').value = out;
}

function saveUserUI() {
  const btn = document.getElementById('userSaveBtn');
  if (btn.disabled) return;

  const errorBox = document.getElementById('userError');
  errorBox.textContent = '';

  const isEdit = !!editingUsername;

  const data = {
    mode: isEdit ? 'edit' : 'create',
    username: document.getElementById('userUsername').value.trim(),
    role: document.getElementById('userRole').value,
    linkedId: document.getElementById('userLinked').value,
    name: document.getElementById('userDisplayName').value.trim(),
    status: isEdit ? document.getElementById('userStatus').value : 'نشط',
    password: document.getElementById('userPassword').value
  };

  if (!isEdit && data.password.length < 8) {
    errorBox.textContent = 'كلمة المرور لازم تكون 8 حروف على الأقل';
    return;
  }

  btn.disabled = true;
  btn.textContent = 'جاري الحفظ...';

  server
    .withSuccessHandler(function() {
      btn.disabled = false;
      btn.textContent = '💾 حفظ';
      closeUserModal();
      loadUsers();

      if (!isEdit) {
        alert(
          'تم إنشاء الحساب ✅\n\n' +
          'اسم المستخدم: ' + data.username.toLowerCase() + '\n' +
          'كلمة المرور: ' + data.password + '\n\n' +
          'سلّمها للمستخدم. لن تظهر كلمة المرور مرة أخرى.'
        );
      } else if (data.password) {
        alert('تم حفظ التعديلات وتغيير كلمة المرور ✅');
      }
    })
    .withFailureHandler(function(error) {
      btn.disabled = false;
      btn.textContent = '💾 حفظ';
      errorBox.textContent =
        String(error.message || error).replace(/^Error:\s*/, '');
    })
    .saveUser(data);
}

function deleteUserUI(username) {
  if (!confirm(
    'هل أنت متأكد من حذف الحساب "' + username + '"؟\n' +
    'لن يستطيع المستخدم الدخول بعد الآن.'
  )) {
    return;
  }

  server
    .withSuccessHandler(function() {
      loadUsers();
    })
    .withFailureHandler(function(error) {
      alert('حدث خطأ: ' + String(error.message || error));
    })
    .deleteUser(username);
}

        document.querySelector(
          '.menu-item[onclick^="showTeachers"]'
        );

      if (item) {
        showTeachers(item);
      }

      break;

    case 'lessons': {

      const item =
        document.querySelector(
          '.menu-item[onclick^="showLessons"]'
        );

      if (item) {
        showLessons(item);
      }

      break;
    }


    case 'subscriptions': {

      const item =
        document.querySelector(
          '.menu-item[onclick^="showSubscriptions"]'
        );

      if (item) {
        showSubscriptions(item);
      }

      break;
    }


    case 'salaries': {

      const item =
        document.querySelector(
          '.menu-item[onclick^="showSalaries"]'
        );

      if (item) {
        showSalaries(item);
      }

      break;
    }


        case 'myreports': {
          const item =
            document.querySelector(
              '.menu-item[onclick^="showMyReports"]'
            );
          if (item) {
            showMyReports(item);
          }

          '.menu-item[onclick^="showUsers"]'

        showUsers(item);

        document.querySelector(
          '.menu-item[onclick^="showDashboard"]'
        );

      if (item) {
        showDashboard(item);
      }

      break;
    }

/* =========================
   تنسيق تاريخ التقرير
========================= */

function formatReportDate(value) {

  if (!value) return 'غير محدد';

  const date = new Date(value);

  if (isNaN(date.getTime())) {
    return String(value);
  }

  return date.toLocaleDateString('ar-EG', {
    year: 'numeric',
    month: 'long',
    day: 'numeric'
  });
}

/* ======================
   تعديل طالب
====================== */

function editStudent(id) {

  const student =
    studentsData.find(function(item) {

      return String(item.id).trim() ===
             String(id).trim();

    });


  if (!student) {

    alert('لم يتم العثور على بيانات الطالب.');

    return;

  }


  /*
   * ==========================
   * فتح نموذج الطالب
   * ==========================
   */

  const modal =
    document.getElementById('studentModal');

  if (!modal) {

    alert('لم يتم العثور على نافذة الطالب.');

    return;

  }


  /*
   * التأكد من أن نموذج الطالب موجود
   *
   * لو كان Modal آخر قد استخدم studentModal
   * نقوم بإعادة تحميل صفحة إضافة الطالب
   */

  if (
    !document.getElementById('studentName') ||
    !document.getElementById('lessonTime')
  ) {

    clearStudentForm();

    modal.style.display = 'flex';

    /*
     * في حالة أن النموذج الأصلي غير موجود
     * نغلق ونفتح نموذج الطالب بالطريقة الأصلية.
     */

    modal.style.display = 'none';

    openAddStudent();

  } else {

    modal.style.display = 'flex';

  }


  /*
   * ==========================
   * البيانات الأساسية
   * ==========================
   */

  document.getElementById('studentId').value =
    student.id || '';


  document.getElementById('studentName').value =
    student.name || '';


  document.getElementById('whatsapp').value =
    student.whatsapp || '';


  document.getElementById('birthDate').value =
    student.birthDate || '';


  document.getElementById('program').value =
    student.program || '';


  document.getElementById('level').value =
    student.level || '';


  document.getElementById('supervisor').value =
    student.supervisor || '';


  document.getElementById('monthlyLessons').value =
    student.monthlyLessons || '8';

  document.getElementById('subscriptionPrice').value =
    student.subscriptionPrice || '';
  
  document.getElementById('lessonDuration').value =
    student.lessonDuration || '30';


  document.getElementById('lessonTime').value =
    student.lessonTime || '';


  document.getElementById('lessonLink').value =
    student.lessonLink || '';


  document.getElementById('subscriptionStart').value =
    student.subscriptionStart || '';


  document.getElementById('status').value =
    student.status || 'نشط';


  document.getElementById('notes').value =
    student.notes || '';


  /*
   * ==========================
   * أيام الحصص
   * ==========================
   */

  const selectedDays =
    String(student.lessonDays || '')
      .split(',')
      .map(function(day) {

        return String(day).trim();

      })
      .filter(function(day) {

        return day !== '';

      });


  document
    .querySelectorAll('.lesson-day')
    .forEach(function(checkbox) {

      checkbox.checked =
        selectedDays.includes(
          String(checkbox.value)
        );

    });


  /*
   * ==========================
   * تحميل المدرسين
   * ==========================
   */

  loadTeacherOptions();


  /*
   * بعد تحميل المدرسين
   * نختار المدرس الحالي
   */

  setTimeout(function() {

    const teacherSelect =
      document.getElementById('teacher');

    if (teacherSelect) {

      /*
       * student.teacher في بيانات الطالب
       * يحتوي غالبًا على اسم المدرس.
       *
       * نحاول أولًا مطابقة القيمة مباشرة،
       * ثم نبحث بالاسم داخل الخيارات.
       */

      teacherSelect.value =
        student.teacher || '';


      if (
        teacherSelect.value !==
        String(student.teacher || '')
      ) {

        Array.from(
          teacherSelect.options
        ).forEach(function(option) {

          if (
            String(option.textContent).trim() ===
            String(student.teacher || '').trim()
          ) {

            teacherSelect.value =
              option.value;

          }

        });

      }

    }

  }, 500);


  /*
   * ==========================
   * عنوان النافذة
   * ==========================
   */

  document.getElementById('modalTitle').textContent =
    'تعديل بيانات الطالب';


  /*
   * إظهار النافذة
   */

  modal.style.display = 'flex';

}



/* ======================
   حفظ
====================== */

function saveStudent() {

  const selectedDays =
    Array.from(
      document.querySelectorAll('.lesson-day:checked')
    ).map(function(checkbox) {
      return checkbox.value;
    });


  const student = {

    id:
      document.getElementById('studentId').value,

    name:
      document.getElementById('studentName').value,


    whatsapp:
      document.getElementById('whatsapp').value,


    birthDate:
      document.getElementById('birthDate').value,

    program:
      document.getElementById('program').value,

    level:
      document.getElementById('level').value,

    teacher:
      document.getElementById('teacher').value,

    supervisor:
      document.getElementById('supervisor').value,

    monthlyLessons:
      document.getElementById('monthlyLessons').value,

    subscriptionPrice:
      document.getElementById('subscriptionPrice').value,
    
    lessonDuration:
      document.getElementById('lessonDuration').value,

    lessonDays:
      selectedDays.join(','),

    lessonTime:
      document.getElementById('lessonTime').value,

    lessonLink:
      document.getElementById('lessonLink').value,

    subscriptionStart:
      document.getElementById('subscriptionStart').value,

    status:
      document.getElementById('status').value,

    notes:
      document.getElementById('notes').value
  };


  if (!student.name.trim()) {

    alert('من فضلك اكتب اسم الطالب');

    return;
  }


  if (!student.teacher) {

    alert('من فضلك اختر المدرس');

    return;
  }


  if (selectedDays.length === 0) {

    alert('من فضلك اختر يومًا واحدًا على الأقل للحصص');

    return;
  }


  if (!student.lessonTime) {

    alert('من فضلك اختر وقت الحصة');

    return;
  }


  if (!student.subscriptionStart) {

    alert('من فضلك اختر تاريخ بداية الاشتراك');

    return;
  }


  const saveButton =
    document.querySelector(
      '#studentModal .btn-primary'
    );


  if (saveButton) {

    saveButton.disabled = true;

    saveButton.innerText =
      'جاري حفظ الطالب وإنشاء الجدول...';

  }


  server

    .withSuccessHandler(function(result) {

      if (saveButton) {

        saveButton.disabled = false;

        saveButton.innerText =
          '💾 حفظ الطالب';

      }

      alert(result.message);

      closeStudentModal();

      // تحديث قائمة الطلاب
      loadStudents();

      // تحديث الحصص بعد تعديل بيانات الطالب
      if (typeof loadLessons === 'function') {
        loadLessons();
      }

      // تحديث الاشتراكات بعد تعديل السعر أو عدد الحصص
      if (typeof loadSubscriptions === 'function') {
        loadSubscriptions();
      }

    })

    .withFailureHandler(function(error) {

      if (saveButton) {

        saveButton.disabled = false;

        saveButton.innerText =
          '💾 حفظ الطالب';

      }


      showError(error);

    })

    .saveStudent(student);

}

function removeStudent(id) {

  const student =
    studentsData.find(function(item) {

      return String(item.id) === String(id);

    });


  if (!student) return;


  const confirmed =
    confirm(
      'هل أنت متأكد من حذف الطالب: ' +
      student.name +
      ' ؟'
    );


  if (!confirmed) return;


  server

    .withSuccessHandler(function(result) {

      alert(result.message);

      loadStudents();

      if (typeof loadLessons === 'function') {
        loadLessons();
      }

      if (typeof loadSubscriptions === 'function') {
        loadSubscriptions();
      }

    })

    .withFailureHandler(showError)

    .deleteStudent(id);

}

/* إغلاق النموذج */
function getSystemModal(modalId) {

  let modal = document.getElementById(modalId);

  if (modal) {
    return modal;
  }

  modal = document.createElement('div');

  modal.id = modalId;
  modal.className = 'modal';

  modal.style.display = 'none';

  document.body.appendChild(modal);

  return modal;
}


let activeSystemModal = null;


function closeModal() {

  if (activeSystemModal) {

    activeSystemModal.style.display = 'none';
    activeSystemModal.innerHTML = '';

    activeSystemModal = null;

    return;
  }

  const modal = document.getElementById('studentModal');

  if (modal) {
    modal.style.display = 'none';
  }
}


function openSystemModal(modalId, html) {

  const modal = getSystemModal(modalId);

  modal.innerHTML = html;

  modal.style.display = 'flex';

  activeSystemModal = modal;

  return modal;
}

document.addEventListener('click', function (event) {
  const modal = document.getElementById('studentModal');

  if (modal && event.target === modal) {

  }
});

/* ======================
   تاريخ اليوم
====================== */

function getTodayString() {

  const now = new Date();

  const year =
    now.getFullYear();

  const month =
    String(now.getMonth() + 1)
      .padStart(2, '0');

  const day =
    String(now.getDate())
      .padStart(2, '0');

  return `${year}-${month}-${day}`;

}


/* ======================
   تحميل حصص التاريخ المختار
====================== */

function loadLessonsForSelectedDate() {

  const dateInput =
    document.getElementById('lessonDateFilter');

  if (!dateInput) return;

  const selectedDate =
    dateInput.value;

  const container =
    document.getElementById('lessonsTable');

  if (!selectedDate) {

    if (container) {

      container.innerHTML = `
        <div class="empty">

          <div style="font-size:40px">
            📅
          </div>

          <h3>
            اختر تاريخًا
          </h3>

          <p>
            اختر التاريخ لعرض حصص هذا اليوم.
          </p>

        </div>
      `;

    }

    return;
  }


  if (container) {

    container.innerHTML = `
      <div class="loading">
        جاري تحميل حصص
        ${escapeHtml(selectedDate)}
        ...
      </div>
    `;

  }


  server

    .withSuccessHandler(function(data) {

      allLessons = data || [];


      const filtered =
        allLessons.filter(function(lesson) {

          let lessonDate =
            String(lesson.date || '').trim();

          if (lessonDate.length >= 10) {

            lessonDate =
              lessonDate.substring(0, 10);

          }

          return lessonDate === selectedDate;

        });


      /* ترتيب الحصص حسب الوقت */

      filtered.sort(function(a, b) {

        return convertTimeToMinutes(a.time)
             - convertTimeToMinutes(b.time);

      });


      renderLessons(filtered);

    })

    .withFailureHandler(function(error) {

      if (container) {

        container.innerHTML = `

          <div class="empty">

            <div style="font-size:40px">
              ⚠️
            </div>

            <h3>
              حدث خطأ أثناء تحميل الحصص
            </h3>

            <p>
              ${escapeHtml(
                error.message || String(error)
              )}
            </p>

          </div>

        `;

      }

    })

    .getLessons();

}

/* ======================
   فلترة الحصص
====================== */

function filterLessons() {

  const searchInput = document.getElementById('lessonSearch');
  const dateInput = document.getElementById('lessonDateFilter');

  const search = searchInput
    ? searchInput.value.toLowerCase().trim()
    : '';

  const selectedDate = dateInput
    ? dateInput.value
    : '';


  const filtered = allLessons.filter(function(lesson) {

    // البحث
    const matchesSearch =
      !search ||
      String(lesson.id || '').toLowerCase().includes(search) ||
      String(lesson.studentName || '').toLowerCase().includes(search) ||
      String(lesson.studentId || '').toLowerCase().includes(search) ||
      String(lesson.teacherName || '').toLowerCase().includes(search) ||
      String(lesson.teacherId || '').toLowerCase().includes(search);


    // التاريخ
    let matchesDate = true;

    if (selectedDate) {

      let lessonDate = String(lesson.date || '').trim();

      /*
       * التأكد أن التاريخ بالشكل:
       * yyyy-mm-dd
       */
      if (lessonDate.length >= 10) {
        lessonDate = lessonDate.substring(0, 10);
      }

      matchesDate = lessonDate === selectedDate;
    }


    return matchesSearch && matchesDate;

  });


  renderLessons(filtered);
}

function updateSubscriptionStats(data) {

  const total = data.length;

  const active = data.filter(function(item) {
    return Number(item.remainingLessons) > 0;
  }).length;

  const finished = data.filter(function(item) {
    return Number(item.remainingLessons) === 0;
  }).length;

  const unpaid = data.filter(function(item) {
    return item.isPaid !== true;
  }).length;

  const remaining = data.reduce(function(sum, item) {
    return sum + (Number(item.remainingLessons) || 0);
  }, 0);

  function setText(id, value) {
    const el = document.getElementById(id);
    if (el) el.textContent = value;
  }

  setText('subscriptionsTotal', total);
  setText('subscriptionsActive', active);
  setText('subscriptionsRemaining', remaining);
  setText('subscriptionsFinished', finished);
  setText('subscriptionsUnpaid', unpaid);
}

function filterSubscriptions() {

  const input =
    document.getElementById(
      'subscriptionSearch'
    );

  if (!input) return;

  const search =
    input.value
      .trim()
      .toLowerCase();

  if (!search) {

    renderSubscriptions(
      subscriptionsData
    );

    return;
  }

  const filtered =
    subscriptionsData.filter(function(item) {

      return (

        String(item.id || '')
          .toLowerCase()
          .includes(search)

        ||

        String(item.studentId || '')
          .toLowerCase()
          .includes(search)

        ||

        String(item.studentName || '')
          .toLowerCase()
          .includes(search)

      );

    });

  renderSubscriptions(filtered);

}


  
function convertTimeToMinutes(time) {

  if (!time) {
    return 99999;
  }


  const text =
    String(time).trim().toUpperCase();


  const match =
    text.match(/(\d{1,2}):(\d{2})\s*(AM|PM)?/);


  if (!match) {
    return 99999;
  }


  let hour =
    parseInt(match[1], 10);

  const minute =
    parseInt(match[2], 10);

  const period =
    match[3];


  if (period === 'PM' && hour !== 12) {

    hour += 12;

  }


  if (period === 'AM' && hour === 12) {

    hour = 0;

  }


  return hour * 60 + minute;

}

function escapeHtml(value) {

  if (value === null || value === undefined) {
    return '';
  }

  return String(value)

    .replace(/&/g, '&amp;')

    .replace(/</g, '&lt;')

    .replace(/>/g, '&gt;')

    .replace(/"/g, '&quot;')

    .replace(/'/g, '&#039;');

}

const DAY_NAMES = ['الأحد','الإثنين','الثلاثاء','الأربعاء','الخميس','الجمعة','السبت'];

function formatLessonDays(value) {
  const parts = String(value ?? '')
    .split(',')
    .map(function(s) { return s.trim(); })
    .filter(function(s) { return s !== ''; });

  if (!parts.length) return '-';

  return parts.map(function(p) {
    return DAY_NAMES[Number(p)] || p;
  }).join('، ');
}

function formatLessonTime(value) {
  const m = String(value || '').match(/^(\d{1,2}):(\d{2})/);
  if (!m) return value || '-';

  let h = Number(m[1]);
  const ampm = h >= 12 ? 'PM' : 'AM';
  h = h % 12 || 12;

  return h + ':' + m[2] + ':00 ' + ampm;
}

  function toHHmm_(text) {
    var s = String(text || '').trim();
    var m = s.match(/(\d{1,2}):(\d{2})\s*(AM|PM|ص|م)?/i);
    if (!m) return '';
    var h = Number(m[1]);
    var suf = (m[3] || '').toUpperCase();
    if ((suf === 'PM' || suf === 'م') && h < 12) h += 12;

  /* توزيع الحصص المتداخلة جنب بعضها داخل اليوم */
  function layoutDay_(events) {
    events.sort(function (a, b) { return a.s - b.s || b.e - a.e; });
 
    var cluster = [], colsEnd = [], clusterEnd = -1;
 
    function flush() {
      cluster.forEach(function (ev) { ev.n = colsEnd.length; });
      cluster = []; colsEnd = []; clusterEnd = -1;
    }
 
    events.forEach(function (ev) {
      if (cluster.length && ev.s >= clusterEnd) flush();
 
      var placed = false;
      for (var c = 0; c < colsEnd.length; c++) {
        if (colsEnd[c] <= ev.s) { ev.c = c; colsEnd[c] = ev.e; placed = true; break; }
      }
      if (!placed) { ev.c = colsEnd.length; colsEnd.push(ev.e); }
 
      cluster.push(ev);
      clusterEnd = Math.max(clusterEnd, ev.e);
    });
    flush();
  }
 
  function renderSchedule_() {
    var scroll = el_('schScroll');
    if (!scroll) return;
 
    var prevTop = scroll.scrollTop;
    var colors = teacherColors_();
 
    /* أيام الأسبوع */
    var days = [];
    for (var i = 0; i < 7; i++) {
      var d = new Date(SCH.weekStart);
      d.setDate(d.getDate() + i);
      days.push({ date: d, key: dateKey_(d), events: [] });
    }
    var byKey = {};
    days.forEach(function (dy) { byKey[dy.key] = dy; });
 
    /* حصص الأسبوع */
    var count = 0, minStart = 24 * 60, maxEnd = 0;
 
    SCH.lessons.forEach(function (l) {
      var dk = String(l.date || '').substring(0, 10);
      var day = byKey[dk];
      if (!day) return;
      if (SCH.teacher && String(l.teacherId) !== SCH.teacher) return;
 
      var t = toHHmm_(l.time);
      if (!t) return;
 
      var tp = t.split(':').map(Number);
      var s = tp[0] * 60 + tp[1];
      var dur = Number(l.duration) || 30;
 
      day.events.push({ lesson: l, s: s, e: s + dur, c: 0, n: 1 });
      minStart = Math.min(minStart, s);
      maxEnd = Math.max(maxEnd, s + dur);
      count++;
    });
 
    var minHour = Math.max(0, Math.min(8, Math.floor(minStart / 60)));
    var maxHour = Math.min(24, Math.max(22, Math.ceil(maxEnd / 60)));
    if (!count) { minHour = 8; maxHour = 22; }
 
    var totalH = (maxHour - minHour) * HOUR_H;
    var todayKey = todayKeyServer_();
 
    /* رأس الأيام */
    var head = '<div class="sch-row sch-head"><div class="sch-corner"></div>';
    days.forEach(function (dy, idx) {
      head += '<div class="sch-dayhead ' + (dy.key === todayKey ? 'today' : '') + '">' +
                '<div class="dn">' + DAY_NAMES[idx] + '</div>' +
                '<div class="dd">' + dy.date.getDate() + '</div></div>';
    });
    head += '</div>';
 
    /* عمود الساعات */
    var gutter = '<div class="sch-gutter" style="height:' + totalH + 'px;">';
    for (var h = minHour; h < maxHour; h++) {
      gutter += '<div class="sch-hour" style="top:' + ((h - minHour) * HOUR_H) + 'px;">' +
                (h === 0 ? '' : fmt12_(h, 0)) + '</div>';
    }
    gutter += '</div>';
 
    /* أعمدة الأيام */
    var lineBg = 'background-image:repeating-linear-gradient(to bottom,transparent 0,transparent ' +
                 (HOUR_H - 1) + 'px,#eee ' + (HOUR_H - 1) + 'px,#eee ' + HOUR_H + 'px);';
    var nowMin = null;
    var nowD = new Date(wallNow_());
    nowMin = nowD.getUTCHours() * 60 + nowD.getUTCMinutes();
 
    var cols = '';
    days.forEach(function (dy) {
      layoutDay_(dy.events);
 
      var evHtml = '';
      dy.events.forEach(function (ev) {
        var l = ev.lesson;
        var st = effStatus_(l);
        var top = (ev.s - minHour * 60) / 60 * HOUR_H;
        var height = Math.max((ev.e - ev.s) / 60 * HOUR_H - 2, 24);
        var width = 100 / ev.n;
        var right = ev.c * width;
 
        var col = (colors[String(l.teacherId)] || { color: PALETTE[0] }).color;
        var sh = Math.floor(ev.s / 60), sm = ev.s % 60;
        var eh = Math.floor((ev.e % 1440) / 60), em = ev.e % 60;
 
        var line2 = role_() === 'admin' ? l.teacherName : STATUS_TEXT[st];
        var line3 = role_() === 'admin' ? STATUS_TEXT[st] : '';
        var tall = height >= 44;
 
        evHtml +=
          '<div class="sch-ev s-' + st + '" data-id="' + escapeHtml(l.id) + '" ' +
            'title="' + escapeHtml(l.studentName + ' — ' + fmt12_(sh, sm) + ' إلى ' + fmt12_(eh, em)) + '" ' +
            'style="top:' + top + 'px;height:' + height + 'px;right:' + right + '%;width:calc(' + width + '% - 2px);' +
            'background:' + col + '22;border-right-color:' + col + ';">' +
            '<div class="t">' + (st === 'done' ? '✓ ' : '') + escapeHtml(l.studentName) + '</div>' +
            (tall ? '<div class="s">' + fmt12_(sh, sm) + ' – ' + fmt12_(eh, em) + '</div>' : '') +
            (height >= 62 ? '<div class="s">' + escapeHtml(line2) + '</div>' : '') +
            (height >= 80 && line3 ? '<div class="s">' + escapeHtml(line3) + '</div>' : '') +
          '</div>';
      });
 
      var nowLine = '';
      if (dy.key === todayKey && nowMin >= minHour * 60 && nowMin <= maxHour * 60) {
        nowLine = '<div class="sch-now" style="top:' + ((nowMin - minHour * 60) / 60 * HOUR_H) + 'px;"></div>';
      }
 
      cols += '<div class="sch-col ' + (dy.key === todayKey ? 'today' : '') + '" ' +
              'style="height:' + totalH + 'px;' + lineBg + '">' + evHtml + nowLine + '</div>';
    });
 
    scroll.innerHTML = head + '<div class="sch-row">' + gutter + cols + '</div>';
 
    scroll.querySelectorAll('.sch-ev').forEach(function (node) {

    });

    if (!SCH.scrolled) {

      scroll.scrollTop = Math.max(0, (focus - minHour * 60) / 60 * HOUR_H - 30);
      SCH.scrolled = true;
    } else {
      scroll.scrollTop = prevTop;
    }
 
    /* الشريط العلوي */
    var a = days[0].date, b = days[6].date;
    el_('schRange').textContent =
      'السبت ' + a.getDate() + ' ' + MONTH_NAMES[a.getMonth()] +
      ' – الجمعة ' + b.getDate() + ' ' + MONTH_NAMES[b.getMonth()] + ' ' + b.getFullYear();
    el_('schPick').value = dateKey_(SCH.weekStart);
    el_('schMeta').textContent = count + ' حصة هذا الأسبوع' +
      (SCH.updatedAt ? ' • آخر تحديث ' + SCH.updatedAt : '');
 
    /* دليل الألوان */
    var legend = '';
    if (role_() === 'admin') {
      Object.keys(colors).forEach(function (id) {
        legend += '<span><i style="background:' + colors[id].color + '"></i>' +
                  escapeHtml(colors[id].name) + '</span>';
      });
      legend += '<span style="color:#8a8076;">|</span>';
    }
    legend += '<span><i style="background:#f8d7da;border:1px solid #b02a37"></i>الطالب لم يحضر</span>' +
              '<span><i style="background:#fff3cd;border:1px solid #d4a017"></i>الطالب إعتذر</span>' +
              '<span><i style="background:#e7f5ec;border:1px solid #287a4b"></i>✓ مكتملة</span>';

  (function addMenuItem_() {
    if (document.querySelector('.sidebar .menu-item[onclick^="showSchedule"]')) return;
 
    var anchor = document.querySelector('.sidebar .menu-item[onclick^="showLessons"]');
    if (!anchor) return;
 
    var item = document.createElement('div');
    item.className = 'menu-item';
    item.setAttribute('onclick', 'showSchedule(this)');
    item.textContent = '🗓️ الجدول';
    anchor.insertAdjacentElement('afterend', item);
 
    MENU_PAGE_BY_HANDLER.showSchedule = 'schedule';
    if (ROLE_PAGES.teacher.indexOf('schedule') === -1) ROLE_PAGES.teacher.push('schedule');
 
    if (role_() === 'student') item.style.display = 'none';
  })();

    if (getCurrentPage() === 'schedule' && (role_() === 'admin' || role_() === 'teacher')) {
      var item = document.querySelector('.sidebar .menu-item[onclick^="showSchedule"]');
      if (item) { window.showSchedule(item); return; }
    }
    return prevRestore.apply(this, arguments);
  };
